<objective>
Fix puzzle image loading for non-NLI libraries (Manchester, Oxford, JTS, Cambridge).

Currently, adding a Manchester, Oxford, or JTS manuscript to the puzzle fails with "No images found" because the image resolution paths only understand NLI IIIF FL IDs. These libraries serve images via IIIF canvas URLs stored in images_ext (populated by enrich_metadata), not FL IDs.

Purpose: Researchers need to assemble fragments from any Genizah holding library, not just NLI-hosted ones.
Output: Desktop and web puzzle both load images for Manchester/Oxford/JTS using images_ext canvas URLs when NLI FL IDs are not available.
</objective>

```python
@app.get('/api/puzzle_folios/{sys_id}')
def puzzle_folios(sys_id: str):
    """Get ordered folio list for a manuscript (for prev/next navigation).

    For NLI-hosted manuscripts: uses IIIF manifest FL IDs.
    For external libraries (Manchester, Oxford, JTS): uses enrich_metadata images_ext.
    """
    # Try NLI IIIF manifest first
    fl_ids = fetch_fl_ids_from_nli(sys_id)
    if fl_ids:
        result = []
        for i, fid in enumerate(fl_ids):
            leaf = (i // 2) + 1
            side = 'r' if i % 2 == 0 else 'v'
            result.append({'fl_id': fid, 'label': f'{leaf}{side}', 'image_url': ''})
        return result

    # Fallback: use enrich_metadata images_ext (Manchester, Oxford, JTS, Cambridge)
    if not state.meta_mgr:
        return []
    try:
        data = state.meta_mgr.enrich_metadata(sys_id)
        images_ext = (data or {}).get('images_ext', [])
        external_provider = (data or {}).get('external_provider', '')
        if images_ext:
            result = []
            for i, img in enumerate(images_ext):
                label = img.get('label', '') or str(i + 1)
                canvas_url = img.get('url', '')
                result.append({
                    'fl_id': '',
                    'label': label,
                    'image_url': canvas_url,
                    'page_index': i,
                    'external_provider': external_provider,
                })
            return result
    except Exception as e:
        logger.warning(f"puzzle_folios fallback failed for {sys_id}: {e}")
    return []
```
