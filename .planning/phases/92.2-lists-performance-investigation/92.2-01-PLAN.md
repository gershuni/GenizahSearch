            latencies = sorted(_inst_query_latencies_ms)
            n = len(latencies)
            def _pct(p: float) -> float:
                if n == 0:
                    return 0.0

                return float(latencies[idx])
            return {
                'query_count': _inst_query_count,
                'client_build_count': _inst_client_build_count,
                'p50_query_latency_ms': _pct(0.50),

                snap = _inst_snapshot()
