        now = _time.time() if now is None else now
        if NLI_DISK_CACHE_TTL <= 0:

        entries: dict[str, dict] = {}
        for system_id, fl_ids in cache.items():

            if not isinstance(cached_at, (int, float)):

            normalized_fl_ids = _normalize_persisted_fl_ids(fl_ids)
            if not normalized_fl_ids:

                'fl_ids': normalized_fl_ids,
                'cached_at': float(cached_at),

        temp_path = f"{cache_path}.tmp.{os.getpid()}.{threading.get_ident()}"

        with _nli_persist_lock:
            success = False
            try:
                cache_dir = os.path.dirname(cache_path)
                if cache_dir:
                    os.makedirs(cache_dir, exist_ok=True)
                with open(temp_path, 'w', encoding='utf-8') as handle:
                    json.dump(payload, handle, ensure_ascii=True, sort_keys=True)

                # Retry os.replace on EACCES / WinError 5 (Windows file-in-use).
                # Delays per D-NLI-01: 100/250/500ms (attempt 0 = immediate; then 3 retries).
                last_exc: Exception | None = None
                for attempt, delay_ms in enumerate((0, 100, 250, 500)):
                    if delay_ms:
                        _time.sleep(delay_ms / 1000.0)
                    try:
                        os.replace(temp_path, cache_path)
                        success = True
                        return True
                    except (PermissionError, OSError) as e:

                        if eno not in (errno.EACCES, errno.EPERM, 13):
                            # Not the WinError-5 retry case -- exit retry loop and
                            # fall through to the outer warning + cleanup.
                            last_exc = e

                logger.warning(

                    "attempts (last error: %s)",

                return False
