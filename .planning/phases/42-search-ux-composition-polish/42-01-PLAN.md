Add search progress instrumentation to all search modes across both apps: elapsed timer, ETA for composition search, chunk count display, summary line after completion, and min-chunks filter for regular chunk search.

Output: Both web and desktop apps show elapsed time during searches, ETA during composition/parallels, chunk count, post-search summary, and a min-chunks filter in parallels for regular mode.

   ```python
   def on_comp_progress(self, curr, total):
       if total:
           self.comp_progress.setRange(0, total)
       else:
           self.comp_progress.setRange(0, 0)
       self.comp_progress.setValue(curr)
       self.comp_chunks_processed = curr
       self.comp_chunks_total = total

       elapsed = time.time() - self.comp_search_start_time
       elapsed_str = f"{int(elapsed // 60)}:{int(elapsed % 60):02d}"

       # Compute ETA with smoothing (update every 2s)
       eta_str = ""
       now = time.time()
       if curr > 0 and total > 0 and (now - self.comp_last_eta_update) >= 2.0:
           rate = curr / elapsed
           remaining = (total - curr) / rate
           eta_str = f" — ~{int(remaining // 60)}:{int(remaining % 60):02d} remaining"
           self.comp_last_eta_text = eta_str
           self.comp_last_eta_update = now
       elif self.comp_last_eta_text:
           eta_str = self.comp_last_eta_text

       self.comp_progress.setFormat(f"{elapsed_str} — {curr}/{total} chunks{eta_str}")
   ```
