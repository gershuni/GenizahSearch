**Ambiguity policy (Codex HIGH #2 — NEW):**
Build a transient builder dict `index_builder: Dict[str, Set[Tuple[str, str]]]` keyed by
normalized key, value a set of `(sys_id, shelfmark)` tuples. After the walk, the runtime
index includes ONLY keys whose set contains exactly ONE distinct sys_id (across however
many shelfmark variants). All other keys are "ambiguous" and get written to
`reports/cudl_alias_collisions.csv` for diagnostic purposes.

Rationale: same sys_id contributing two variants to the same normalized key is fine (no
ambiguity — both point to one row). Different sys_ids contributing variants to the same
normalized key IS ambiguity (we cannot decide which row the CUDL classmark should resolve
to without more context, so we exclude the key entirely — fail-loud over silent wrong-answer).

   ```python
   from collections import defaultdict

   def build_alias_index(csv_bank):
       """Build the CUDL alias index with strict ambiguity-exclusion (Codex HIGH #2).

       For each CUL/Mosseri row, walk its variants and produce normalized index keys
       via the Mosseri forward path (construct_mosseri_cudl_label) and/or the generic
       CUDL-form path (cudl_normalize). Collect all (sys_id, shelfmark) claims per key
       into a builder set. After the walk, include ONLY keys with exactly one distinct
       sys_id. Write excluded ambiguous keys to reports/cudl_alias_collisions.csv.
       """
       from genizah_core import construct_mosseri_cudl_label  # late import — break cycle
       global _CUDL_ALIAS_INDEX

       # Ensure collision keys (leading-zero) are loaded so we can defend the index
       # against the D-06 risk too. load_collision_keys is fully implemented in Plan 03;
       # if it is still a stub here, this is a no-op — the policy below catches the
       # remaining cases via the per-key sys_id-set check.
       try:
           load_collision_keys()
       except Exception as e:
           logger.debug("load_collision_keys failed (continuing): %s", e)

       def _index_key_for_label(label: str) -> str:
           # Strip the zfill leading zeros that construct_mosseri_cudl_label adds, since
           # CUDL's URL slug form drops them (MS-MOSSERI-III-00027-O -> mosseriiii27o).
           parts = []
           for seg in label.split('-'):
               if seg.isdigit():
                   parts.append(seg.lstrip('0') or '0')
               else:
                   parts.append(seg)
           return cudl_normalize('-'.join(parts))

       index_builder: 'defaultdict[str, set]' = defaultdict(set)

       for sys_id, data in csv_bank.items():
           variants = data.get('call_numbers_raw') or []
           primary = data.get('shelfmark')
           if primary and primary not in variants:
               variants = list(variants) + [primary]

           canonical = data.get('shelfmark') or (variants[0] if variants else '')
           library_code = data.get('library_code') or ''

           for variant in variants:
               # --- Mosseri forward path (D-03) ---
               if library_code == 'Mosseri':
                   label = construct_mosseri_cudl_label(variant)
                   if label:
                       key = _index_key_for_label(label)
                       if key and not _is_collision_key(key):
                           index_builder[key].add((sys_id, canonical))

               # --- Generic CUDL-form path (CUL Or./T-S/Add. — applies in Plan 03 too) ---
               if library_code in ('CUL', 'Mosseri'):
                   key = cudl_normalize(variant)
                   if key and not _is_collision_key(key):
                       index_builder[key].add((sys_id, canonical))

       # Materialize: keep only keys with exactly ONE distinct sys_id.
       final_index: Dict[str, tuple] = {}
       ambiguous: Dict[str, set] = {}
       for key, claim_set in index_builder.items():
           sys_ids = {sid for (sid, _shelf) in claim_set}
           if len(sys_ids) == 1:
               # Pick any (sys_id, shelfmark) — they all share the sys_id.
               final_index[key] = next(iter(claim_set))
           else:
               ambiguous[key] = claim_set

       _CUDL_ALIAS_INDEX = final_index

       # Write ambiguous-key audit report (Codex HIGH #2).
       _write_alias_collision_report(ambiguous)

       logger.warning(
           "alias index built: %d keys, %d ambiguous keys excluded",
           len(final_index), len(ambiguous),
       )


   def _write_alias_collision_report(ambiguous: Dict[str, set]) -> None:
       """Write reports/cudl_alias_collisions.csv with columns key,sys_ids,shelfmarks.

       Always writes the file (header even if empty) so consumers can rely on its existence.
       """
       reports_dir = Path(__file__).resolve().parent.parent / "reports"
       try:
           reports_dir.mkdir(exist_ok=True)
       except Exception as e:
           logger.warning("could not create reports dir: %s", e)
           return
       out_path = reports_dir / "cudl_alias_collisions.csv"
       try:
           with out_path.open("w", encoding="utf-8", newline="") as f:
               w = csv.writer(f)
               w.writerow(["key", "sys_ids", "shelfmarks"])
               for key in sorted(ambiguous.keys()):
                   claims = ambiguous[key]
                   sys_ids = sorted({sid for (sid, _s) in claims})
                   shelfmarks = sorted({s for (_sid, s) in claims if s})
                   w.writerow([key, "|".join(sys_ids), "|".join(shelfmarks)])
       except Exception as e:
           logger.warning("failed to write %s: %s", out_path, e)
   ```

"""Phase 84 Plan 02: ambiguity-policy unit tests for shared.shelfmark_bridge.

Codex HIGH #2: build_alias_index must EXCLUDE keys whose normalized form maps to
>1 distinct sys_id, and write them to reports/cudl_alias_collisions.csv.
"""
import csv
from pathlib import Path

import pytest

from shared.shelfmark_bridge import build_alias_index, lookup_cudl


COLLISIONS_REPORT = Path(__file__).resolve().parent.parent / "reports" / "cudl_alias_collisions.csv"


def _read_collisions_keys():
    if not COLLISIONS_REPORT.exists():
        return set()
    with COLLISIONS_REPORT.open("r", encoding="utf-8", newline="") as f:
        return {row["key"] for row in csv.DictReader(f) if row.get("key")}


class TestAmbiguityExclusion:
    def test_two_distinct_sys_ids_same_key_are_excluded(self):
        # Two CUL rows with identical shelfmark → same cudl_normalize key → ambiguity.
        build_alias_index({
            "9A": {"shelfmark": "T-S Y 1", "library_code": "CUL", "call_numbers_raw": ["T-S Y 1"]},
            "9B": {"shelfmark": "T-S Y 1", "library_code": "CUL", "call_numbers_raw": ["T-S Y 1"]},
        })
        assert lookup_cudl("tsy1") is None
        assert "tsy1" in _read_collisions_keys()

    def test_same_sys_id_multiple_paths_is_not_ambiguous(self):
        # One sys_id contributes two equivalent variants → same key → still one sys_id → KEEP.
        build_alias_index({
            "9C": {
                "shelfmark": "Moss. III,27O",
                "library_code": "Mosseri",
                "call_numbers_raw": ["Moss. III,27O", "Moss III 27 O"],
            },
        })
        r = lookup_cudl("mosseriiii27o")
        assert r is not None and r["sys_id"] == "9C"

    def test_collision_report_header(self):
        build_alias_index({
            "9D": {"shelfmark": "T-S Z 1", "library_code": "CUL", "call_numbers_raw": ["T-S Z 1"]},
            "9E": {"shelfmark": "T-S Z 1", "library_code": "CUL", "call_numbers_raw": ["T-S Z 1"]},
        })
        assert COLLISIONS_REPORT.exists()
        first_line = COLLISIONS_REPORT.read_text(encoding="utf-8").splitlines()[0]
        assert first_line == "key,sys_ids,shelfmarks"

    def test_three_way_ambiguity_excluded(self):
        build_alias_index({
            "9F": {"shelfmark": "T-S Q 1", "library_code": "CUL", "call_numbers_raw": ["T-S Q 1"]},
            "9G": {"shelfmark": "T-S Q 1", "library_code": "CUL", "call_numbers_raw": ["T-S Q 1"]},
            "9H": {"shelfmark": "T-S Q 1", "library_code": "CUL", "call_numbers_raw": ["T-S Q 1"]},
        })
        assert lookup_cudl("tsq1") is None
        keys = _read_collisions_keys()
        assert "tsq1" in keys
