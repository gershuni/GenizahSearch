   ```python
   _NUMERIC_RUN_RE = re.compile(r"(\d+)(?:\.(\d+))+")

   def _collapse_numeric_runs(s: str) -> str:
       """Collapse dot-separated numeric runs into a single concatenated number.

       'or1080.1.1' -> 'or1080.11' (NORM-02 numeric-collapse: Or. 1080.1.1 in CUDL)
       'tsar48.211' -> 'tsar48.211' (single numeric segment after dot, unchanged)
       'add863.2'   -> 'add863.2'   (single numeric segment after dot, unchanged)

       Applies AFTER cudl_normalize. Only collapses when there are 2+ dot-separated
       digit groups in a row.
       """
       def _join(m):
           # m.group(0) is the full matched dot-separated digit run (e.g. '1080.1.1' or '48.211')
           parts = m.group(0).split('.')
           # Collapse only when 3+ groups (i.e. 2+ dots inside the run); 2-group runs
           # like '48.211' are kept as-is because CUDL keeps them too.
           if len(parts) >= 3:
               return parts[0] + '.' + ''.join(parts[1:])
           return m.group(0)
       return _NUMERIC_RUN_RE.sub(_join, s)
   ```
