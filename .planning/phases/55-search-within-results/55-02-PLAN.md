    def _enter_refine_mode():

        search_state._refine_mode = True
        search_state._zero_result_refine = False

        ui.run_javascript(f'''

            setTimeout(() => {{

                if (el) {{ var inp = el.querySelector("input"); if (inp) inp.focus(); }}
            }}, 500);
        ''')

    def _exit_refine_mode():
        """D-02a: Cancel refine mode without running search."""
        search_state._refine_mode = False
        refine_badge.set_visibility(False)
        refine_cancel_btn.set_visibility(False)

    def _update_refinement_strip():
        """Rebuild the breadcrumb chip chain (D-04, D-05, D-06, D-07, D-10)."""
        refinement_strip.clear()
        chain = search_state.refinement_chain
        if not chain:
            refinement_strip.style('display: none;')
            return
        refinement_strip.style('display: flex;')
        show_modes = needs_mode_labels(chain)
        with refinement_strip:
            for i, step in enumerate(chain):
                if i > 0:
                    ui.label('\u203a').classes('text-lg mx-1').style('color: #999;')
                label = step.display_label
                if show_modes:
                    label = f"{step.query} ({step.mode})"
                chip = ui.chip(label, removable=True, color='indigo-2').classes('text-sm')
                chip.on('remove', lambda _idx=i: _remove_refinement_step(_idx))
            # Result count for final step only (D-06)
            ui.label(f'{chain[-1].result_count:,}').classes('text-sm font-bold ml-2').style('color: var(--primary-600);')
            # Clear all (D-11)
            ui.button(tr('Clear all'), icon='clear_all', on_click=_clear_refinement_chain).classes('text-xs ml-2').props('flat dense no-caps')
            # Stale indicator (D-16)
            if search_state._refinement_stale:
                ui.label(tr('Scope changed - results will update on next search')).classes('text-xs ml-2').style('color: #e67e22; font-style: italic;')

    def _remove_refinement_step(index):
        """D-12: Remove chip at index and all subsequent, then re-execute."""
        search_state.refinement_chain = truncate_chain(search_state.refinement_chain, index)
        if search_state.refinement_chain:
            # Re-execute chain to rebuild restrict set (D-13)
            _replay_refinement_chain()
        else:
            _clear_refinement_chain()

    def _clear_refinement_chain():
        """D-11: Remove entire chain, return to unrestricted search."""
        search_state.refinement_chain = []
        search_state.refinement_restrict_sys_ids = None
        search_state._refine_mode = False
        search_state._refinement_stale = False
        refine_badge.set_visibility(False)
        refine_cancel_btn.set_visibility(False)
        persist_value('search_refinement_chain', [])
        _update_refinement_strip()
        _update_search_within_btn()

    async def _replay_refinement_chain():
        """Replay chain from stored params to rebuild restrict sets (D-13, D-14)."""
        chain = search_state.refinement_chain
        current_restrict = search_state.restrict_sys_ids  # filter restrict
        for step in chain:
            effective = compute_effective_restrict(current_restrict, None if chain.index(step) == 0 else current_restrict_from_chain)
            # ... sequential re-execution
        # Simplified: replay each step sequentially
        accumulated_restrict = None
        for step in chain:
            effective = compute_effective_restrict(search_state.restrict_sys_ids, accumulated_restrict)
            def _run_step(s=step, r=effective):
                return state.searcher.execute_search(
                    s.query, s.mode, s.gap,
                    exclude_words=s.exclude_words or None,
                    responsa_options=s.responsa_options,
                    restrict_sys_ids=r,
                    text_position=s.text_position,
                )
            step_results = await run.io_bound(_run_step)
            result_ids = {r.get('display', {}).get('id') for r in step_results if r.get('display', {}).get('id')}
            step.result_count = len(result_ids)
            accumulated_restrict = result_ids if result_ids else set()
        search_state.refinement_restrict_sys_ids = accumulated_restrict
        persist_value('search_refinement_chain', [s.to_dict() for s in chain])
        _update_refinement_strip()
        # Re-run the final search to show results
        search_state._refine_mode = False
