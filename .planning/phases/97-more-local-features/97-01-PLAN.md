def _commit_batch(self) -> None:
    if not self._pending_filepaths:
        return
    self._commit_writer_with_retry()  # Tantivy first
    placeholders = ",".join("?" * len(self._pending_filepaths))
    self._conn.execute(
        f"UPDATE processed_files SET status = 'committed' WHERE filepath IN ({placeholders})",

    )
    self._conn.commit()
