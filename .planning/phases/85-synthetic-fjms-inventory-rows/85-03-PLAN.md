class TestParameterizedSqlOnly:
    """T-85-02 mitigation: scan export_fist_enrichment.py for f-string SQL on dynamic values."""

    def test_no_fstring_sql_with_dynamic_values(self):
        path = Path("scripts/export_fist_enrichment.py")
        src = path.read_text(encoding="utf-8")
        # Allow f-strings in print/log statements; block f-strings inside SQL strings.
        # Simple heuristic: find triple-quoted SQL strings that contain f-string interpolation.
        sql_blocks = re.findall(r'""".*?"""', src, re.DOTALL)
        for block in sql_blocks:
            if any(kw in block.upper() for kw in ("SELECT", "INSERT", "UPDATE", "DELETE")):
                # Reject if block contains {variable} interpolation — should use ? params instead.
                # Allow printf inside SQL (sqlite-native) — that's expected.
                bad = re.findall(r"\{[a-zA-Z_][a-zA-Z0-9_]*\}", block)
                # Filter out any false positives that are constants or escaped
                bad = [b for b in bad if b not in ("{}", "{0}")]
                assert not bad, f"f-string interpolation in SQL block: {bad} in {block[:100]}..."

def _create_qualifying_inventories_temp_table(source: sqlite3.Connection, nli_db_path: Optional[Path] = None) -> None:
    """Phase 85 D-02 qualifying set: inventories with no Alma link AND
    (CUDL manifest exists OR substantive FJMS metadata exists).

    Creates `temp.qualifying_synthetic_inventories(InventoryId INTEGER PRIMARY KEY)`.
    Idempotent — drops + recreates on each call.

    Predicate matches Plan 02's `_build_qualifying_inventories` to ensure
    libraries.csv synthetic block and fjms_enrichment.db synthetic AlmaIds
    cover the exact same inventory set.
    """
    source.execute("DROP TABLE IF EXISTS temp.qualifying_synthetic_inventories")
    source.execute("""
        CREATE TEMP TABLE qualifying_synthetic_inventories (
            InventoryId INTEGER PRIMARY KEY
        )
    """)
    # Substantive FJMS metadata: dbo_UnitCatalogRec.Title or .TitleHeb or .GenizahTitle non-empty.
    source.execute("""
        INSERT INTO temp.qualifying_synthetic_inventories(InventoryId)
        SELECT DISTINCT inv.InventoryId
        FROM dbo_Inventory inv
        JOIN dbo_InventorySignature isig ON inv.InventoryId = isig.InventoryId
        JOIN dbo_Signature sig ON isig.SetSignatureId = sig.SetSignatureId
        LEFT JOIN dbo_UnitCatalogRec cat ON sig.SignatureId = cat.SignatureId
        LEFT JOIN dbo_InventoryAlma alma ON alma.InventoryId = inv.InventoryId
        WHERE alma.AlmaId IS NULL
          AND (
              (cat.Title IS NOT NULL AND TRIM(cat.Title) <> '') OR
              (cat.TitleHeb IS NOT NULL AND TRIM(cat.TitleHeb) <> '') OR
              (cat.GenizahTitle IS NOT NULL AND TRIM(cat.GenizahTitle) <> '')
          )
    """)
    # If nli_crossref.db is available, ATTACH it and add CUDL-matching inventories.
    if nli_db_path and Path(nli_db_path).exists():
        source.execute("ATTACH DATABASE ? AS nli", (str(nli_db_path),))
        # Match by normalized shelfmark: SQLite doesn't have cudl_normalize, so we
        # do a coarse pre-filter (lowercase + strip) here and rely on Plan 02's
        # generation script to also exclude any non-matching cases. The temp table
        # is a SUPERSET filter; the synthetic AlmaId UNION block in each export
        # function then naturally aligns with Plan 02's libraries.csv synthetic
        # block (since fjms_service queries are AlmaId-keyed and only AlmaIds
        # that appear in libraries.csv synthetic block are ever queried).
        source.execute("""
            INSERT OR IGNORE INTO temp.qualifying_synthetic_inventories(InventoryId)
            SELECT DISTINCT inv.InventoryId
            FROM dbo_Inventory inv
            JOIN dbo_InventorySignature isig ON inv.InventoryId = isig.InventoryId
            JOIN dbo_Signature sig ON isig.SetSignatureId = sig.SetSignatureId
            LEFT JOIN dbo_InventoryAlma alma ON alma.InventoryId = inv.InventoryId
