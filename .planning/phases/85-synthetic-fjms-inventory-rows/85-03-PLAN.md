def _validate_synthetic_export(target: sqlite3.Connection) -> None:
    """REVIEWS-MODE Codex HIGH: table-specific invariants, NOT global distinct-AlmaId equality.

    Invariants:
      (a) catalog: 1:1 by (AlmaId, UnitCatalogRecId, SignatureId) triple — assert
          no duplicate triples in synthetic block
      (b) For 1:N tables (bibliography, domains, catalog_running_titles, etc.):
          assert every synthetic AlmaId in the table is present in the manifest
          temp table, AND no synthetic AlmaId collides with any real AlmaId
      (c) Cross-table: every synthetic AlmaId must appear in `catalog` at minimum
          (warn — don't fail — on absence)
    """
    from shared.synthetic_sys_id import is_synthetic_sys_id

    # (a) catalog triple uniqueness
    duplicates = target.execute("""
        SELECT AlmaId, UnitCatalogRecId, SignatureId, COUNT(*) as c
        FROM catalog
        WHERE AlmaId LIKE '99%000000'
        GROUP BY AlmaId, UnitCatalogRecId, SignatureId
        HAVING c > 1
    """).fetchall()
    if duplicates:
        raise SystemExit(
            f"REVIEWS-MODE INVARIANT VIOLATION: catalog has duplicate (AlmaId, "
            f"UnitCatalogRecId, SignatureId) triples in synthetic block: {duplicates[:5]}"
        )

    # (b) per-table real/synthetic disjointness
    alma_keyed = ["catalog", "domains", "joins", "catalog_running_titles",
                  "catalog_sizes", "catalog_fields", "catalog_free_desc",
                  "catalog_full_texts", "catalog_textual_frames",
                  "catalog_mentions", "bibliography", "catalog_refs"]
    for table in alma_keyed:
        rows = target.execute(f"SELECT DISTINCT AlmaId FROM {table}").fetchall()
        ids = {r[0] for r in rows}
        synthetic = {a for a in ids if is_synthetic_sys_id(a)}
        real = ids - synthetic
        intersect = synthetic & real
        if intersect:
            raise SystemExit(f"D-01a COLLISION in {table}: {sorted(intersect)[:5]}")
        print(f"  {table}: {len(real)} real, {len(synthetic)} synthetic")

    # (c) Cross-table: synthetic AlmaIds in 1:N tables but NOT in catalog → warn
    catalog_synth = {r[0] for r in target.execute(
        "SELECT DISTINCT AlmaId FROM catalog WHERE AlmaId LIKE '99%000000'"
    )}
    for table in [t for t in alma_keyed if t != "catalog"]:
        try:
            table_synth = {r[0] for r in target.execute(
                f"SELECT DISTINCT AlmaId FROM {table} WHERE AlmaId LIKE '99%000000'"
            )}
        except sqlite3.OperationalError:
            continue
        orphans = table_synth - catalog_synth
        if orphans:
            import sys as _sys
            print(f"  WARNING: {len(orphans)} synthetic AlmaIds in {table} are absent "
                  f"from catalog (e.g. {sorted(orphans)[:3]}) — Phase 86 audit pickup",
                  file=_sys.stderr)
