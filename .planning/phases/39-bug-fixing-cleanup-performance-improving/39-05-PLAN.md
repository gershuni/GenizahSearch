```python
"""E2E test configuration using NiceGUI's Screen fixture."""
import pytest
import os

# Skip all E2E tests if ChromeDriver/Selenium is not available
def pytest_collection_modifyitems(config, items):
    """Skip E2E tests if Selenium/ChromeDriver is not available."""
    try:
        from selenium import webdriver
        from selenium.webdriver.chrome.service import Service
        # Quick check if ChromeDriver is accessible
        # Don't actually start a browser, just verify imports work
    except ImportError:
        skip_marker = pytest.mark.skip(reason="Selenium not installed (pip install selenium)")
        for item in items:
            if "e2e" in str(item.fspath):
                item.add_marker(skip_marker)
        return

    # Check if ChromeDriver binary is accessible
    try:
        from selenium.webdriver.chrome.service import Service as ChromeService
        from webdriver_manager.chrome import ChromeDriverManager
    except ImportError:
        # webdriver_manager not installed - tests will try system ChromeDriver
        pass


@pytest.fixture(scope='module')
def screen():
    """
    NiceGUI Screen fixture for E2E testing.

    Starts a NiceGUI server, provides Selenium browser access,
    and cleans up after tests complete.
    """
    try:
        from nicegui.testing import Screen
        # NiceGUI Screen manages server lifecycle
        s = Screen()
        yield s
        s.stop()
    except Exception as e:
        pytest.skip(f"NiceGUI Screen fixture unavailable: {e}")
```
