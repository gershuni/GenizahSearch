```python
    Starts the server in a subprocess, waits for it to be ready,
    yields the base URL, then terminates the server.
    """
    port = 8099  # Use non-standard port to avoid conflicts
    proc = subprocess.Popen(
        [sys.executable, "-m", "web.main"],
        env={**os.environ, "GENIZAH_PORT": str(port)},
        cwd=str(ROOT_DIR),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
    )

    # Wait for server to start (poll with requests)
    import requests
    base_url = f"http://localhost:{port}"
    for _ in range(30):  # 30 seconds max
        try:
            resp = requests.get(base_url, timeout=1)
            if resp.status_code == 200:
                break
        except requests.ConnectionError:
            time.sleep(1)
    else:
        proc.terminate()
        raise RuntimeError("Web server did not start within 30 seconds")

    yield base_url

    proc.terminate()
    proc.wait(timeout=5)
```

```python
"""Performance and stress tests for GenizahSearch.

These tests validate that the application handles large data sets
without timeouts or memory issues. They don't require a running
web server — they test the underlying search and rendering logic.

Run: pytest tests/test_performance.py -v
"""
import time
import pytest


class TestSearchPerformance:
    """Test search performance with large result sets."""

    def test_search_1000_results_timing(self):
        """Search should complete within 10 seconds even for 1000+ results."""
        try:
            from genizah_core import SearchEngine
            engine = SearchEngine()
        except Exception:
            pytest.skip("SearchEngine not available (Tantivy index missing)")

        start = time.time()
        results = engine.search("א", max_results=1000)  # Common letter
        elapsed = time.time() - start

        assert elapsed < 10.0, f"Search took {elapsed:.1f}s (limit: 10s)"
        # We don't assert exact count since it depends on index content

    def test_result_pagination_slice_performance(self):
        """Slicing large result lists should be instant."""
        # Simulate a large result set
        results = [{'display': {'id': f'sys{i}', 'shelfmark': f'T-S {i}'}, 'snippet': f'text {i}'} for i in range(5000)]

        start = time.time()
        for page in range(100):  # 100 pages of 50
            page_results = results[page * 50: (page + 1) * 50]
            assert len(page_results) == 50
        elapsed = time.time() - start

        assert elapsed < 0.1, f"Pagination slicing took {elapsed:.3f}s (limit: 0.1s)"

    def test_metadata_batch_lookup_performance(self):
        """Batch metadata lookup for search results should complete quickly."""
        try:
            from genizah_core import MetadataManager
            mgr = MetadataManager()
        except Exception:
            pytest.skip("MetadataManager not available")

        # Look up 200 random sys_ids
        sys_ids = list(mgr.csv_bank.keys())[:200] if hasattr(mgr, 'csv_bank') else []
        if not sys_ids:
            pytest.skip("No metadata loaded")

        start = time.time()
        for sid in sys_ids:
            try:
                mgr.get_meta_for_id(sid)
            except Exception:
                pass
        elapsed = time.time() - start

        assert elapsed < 2.0, f"200 metadata lookups took {elapsed:.1f}s (limit: 2s)"


class TestListPerformance:
    """Test list operations with many items."""

    def test_large_list_creation(self):
        """Creating a list with 100+ items should be fast."""
        # Simulate list creation (no actual database)
        items = []
        start = time.time()
        for i in range(200):
            items.append({
                'sys_id': f'sys{i}',
                'shelfmark': f'T-S {i}.{i}',
                'added_at': '2026-01-01',
                'notes': f'Note for item {i}'
            })
        elapsed = time.time() - start

        assert elapsed < 0.01, f"List creation took {elapsed:.4f}s"
        assert len(items) == 200
```
