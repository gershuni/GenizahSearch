```python
@pytest.fixture(scope="session")
def web_server():

    Starts the server in a subprocess, waits for it to be ready,
    yields the base URL, then terminates the server.
    """
    port = 8099  # Use non-standard port to avoid conflicts
    proc = subprocess.Popen(
        [sys.executable, "-m", "web.main"],
        env={**os.environ, "GENIZAH_PORT": str(port)},
        cwd=str(ROOT_DIR),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
    )

    # Wait for server to start (poll with requests)
    import requests
    base_url = f"http://localhost:{port}"
    for _ in range(30):  # 30 seconds max
        try:
            resp = requests.get(base_url, timeout=1)
            if resp.status_code == 200:
                break
        except requests.ConnectionError:
            time.sleep(1)
    else:
        proc.terminate()
        raise RuntimeError("Web server did not start within 30 seconds")

    yield base_url

    proc.terminate()
    proc.wait(timeout=5)
```

```python
"""E2E tests for the search page using Playwright.

Requires: playwright install chromium
Run: pytest tests/test_e2e_search.py --headed (to see browser)
     pytest tests/test_e2e_search.py (headless)

These tests require the web server fixture from conftest.py.
Mark with @pytest.mark.e2e so they can be skipped in unit-test-only runs.
"""
import pytest

pytestmark = pytest.mark.e2e


def test_search_page_loads(web_server, page):
    """Verify the search page loads and shows the search input."""
    page.goto(f"{web_server}/search")
    page.wait_for_load_state("networkidle")

    # Search input should be visible
    search_input = page.locator("input[type='text']").first
    assert search_input.is_visible(), "Search input not found"


def test_search_returns_results(web_server, page):
    """Happy path: enter a search term, verify results appear."""
    page.goto(f"{web_server}/search")
    page.wait_for_load_state("networkidle")

    # Type a common Hebrew search term
    search_input = page.locator("input[type='text']").first
    search_input.fill("שלום")

    # Click search button (or press Enter)
    search_input.press("Enter")

    # Wait for results to appear (results container should have cards)
    page.wait_for_selector(".q-card", timeout=30000)

    # Verify at least one result card exists
    cards = page.locator(".q-card")
    assert cards.count() > 0, "No result cards found after search"


def test_search_result_click_opens_viewer(web_server, page):
    """Click a result card and verify the browse/viewer panel opens."""
    page.goto(f"{web_server}/search")
    page.wait_for_load_state("networkidle")

    search_input = page.locator("input[type='text']").first
    search_input.fill("שלום")
    search_input.press("Enter")

    page.wait_for_selector(".q-card", timeout=30000)

    # Capture URL before click
    url_before = page.url

    # Click first result card
    page.locator(".q-card").first.click()

    # Wait for navigation or content change
    page.wait_for_timeout(3000)

    # Verify: either URL changed (navigated to browse) or new content appeared
    url_after = page.url
    url_changed = url_after != url_before
    # Check for browse-page content indicators (metadata panel, image viewer, shelfmark heading)
    has_browse_content = (
        page.locator("text=/browse/i").count() > 0
        or page.locator("text=/shelfmark/i").count() > 0
        or page.locator("text=/metadata/i").count() > 0
        or "browse" in url_after
    )
    assert url_changed or has_browse_content, (
        f"Clicking result did not navigate or show viewer content. "
        f"URL before={url_before}, after={url_after}"
    )


def test_pagination_controls_appear(web_server, page):
    """Verify pagination controls appear when results exceed one page."""
    page.goto(f"{web_server}/search")
    page.wait_for_load_state("networkidle")

    # Use a broad search term to get many results
    search_input = page.locator("input[type='text']").first
    search_input.fill("א")  # Single common letter — should return many results
    search_input.press("Enter")

    page.wait_for_selector(".q-card", timeout=30000)

    # Check for pagination controls (prev/next buttons)
    # Look for page indicator text like "Page X / Y"
    page.wait_for_timeout(1000)

    # Assert pagination controls are present in the page
    # Look for next-page button (chevron_right icon) or "Page X / Y" text
    page_content = page.content()
    has_page_indicator = bool(
        page.locator("text=/Page \\d+ \\/ \\d+/").count() > 0
        or page.locator("text=/page/i").count() > 0
    )
    has_nav_button = bool(
        page.locator("[aria-label*='next'], [aria-label*='Next']").count() > 0
        or page.locator("button >> text=/chevron_right|arrow_forward|►/").count() > 0
        or page.locator("button:has(i.q-icon)").count() > 0
    )
    assert has_page_indicator or has_nav_button, (
        "Pagination controls not found after broad search. "
        "Expected 'Page X / Y' text or next/prev navigation buttons."
    )
```

```python
"""Performance and stress tests for GenizahSearch.

These tests validate that the application handles large data sets
without timeouts or memory issues. They don't require a running
web server — they test the underlying search and rendering logic.

Run: pytest tests/test_performance.py -v
"""
import time
import pytest


class TestSearchPerformance:
    """Test search performance with large result sets."""

    def test_search_1000_results_timing(self):
        """Search should complete within 10 seconds even for 1000+ results."""
        try:
            from genizah_core import SearchEngine
            engine = SearchEngine()
        except Exception:
            pytest.skip("SearchEngine not available (Tantivy index missing)")

        start = time.time()
        results = engine.search("א", max_results=1000)  # Common letter
        elapsed = time.time() - start

        assert elapsed < 10.0, f"Search took {elapsed:.1f}s (limit: 10s)"
        # We don't assert exact count since it depends on index content

    def test_result_pagination_slice_performance(self):
        """Slicing large result lists should be instant."""
        # Simulate a large result set
        results = [{'display': {'id': f'sys{i}', 'shelfmark': f'T-S {i}'}, 'snippet': f'text {i}'} for i in range(5000)]

        start = time.time()
        for page in range(100):  # 100 pages of 50
            page_results = results[page * 50: (page + 1) * 50]
            assert len(page_results) == 50
        elapsed = time.time() - start

        assert elapsed < 0.1, f"Pagination slicing took {elapsed:.3f}s (limit: 0.1s)"

    def test_metadata_batch_lookup_performance(self):
        """Batch metadata lookup for search results should complete quickly."""
        try:
            from genizah_core import MetadataManager
            mgr = MetadataManager()
        except Exception:
            pytest.skip("MetadataManager not available")

        # Look up 200 random sys_ids
        sys_ids = list(mgr.csv_bank.keys())[:200] if hasattr(mgr, 'csv_bank') else []
        if not sys_ids:
            pytest.skip("No metadata loaded")

        start = time.time()
        for sid in sys_ids:
            try:
                mgr.get_meta_for_id(sid)
            except Exception:
                pass
        elapsed = time.time() - start

        assert elapsed < 2.0, f"200 metadata lookups took {elapsed:.1f}s (limit: 2s)"


class TestListPerformance:
    """Test list operations with many items."""

    def test_large_list_creation(self):
        """Creating a list with 100+ items should be fast."""
        # Simulate list creation (no actual database)
        items = []
        start = time.time()
        for i in range(200):
            items.append({
                'sys_id': f'sys{i}',
                'shelfmark': f'T-S {i}.{i}',
                'added_at': '2026-01-01',
                'notes': f'Note for item {i}'
            })
        elapsed = time.time() - start

        assert elapsed < 0.01, f"List creation took {elapsed:.4f}s"
        assert len(items) == 200
```
