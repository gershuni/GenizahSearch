**Step 3: Add pagination controls to the UI**

After the `render_results` call completes, add a pagination footer inside the results area. Create a helper function `render_pagination_controls(total_results)`:

```python
def render_pagination_controls(total_results):
    """Render prev/next/page-number controls below results."""
    if total_results <= search_state.page_size:
        return  # No pagination needed for small result sets

    total_pages = (total_results + search_state.page_size - 1) // search_state.page_size
    current = search_state.current_page

    with ui.row().classes('w-full justify-center items-center gap-2 py-4'):
        ui.button(icon='first_page', on_click=lambda: go_to_page(0)).props(
            'flat dense' + (' disable' if current == 0 else '')
        )
        ui.button(icon='chevron_left', on_click=lambda: go_to_page(current - 1)).props(
            'flat dense' + (' disable' if current == 0 else '')
        )

        ui.label(f'{tr("Page")} {current + 1} / {total_pages}').classes('mx-4')

        ui.button(icon='chevron_right', on_click=lambda: go_to_page(current + 1)).props(
            'flat dense' + (' disable' if current >= total_pages - 1 else '')
        )
        ui.button(icon='last_page', on_click=lambda: go_to_page(total_pages - 1)).props(
            'flat dense' + (' disable' if current >= total_pages - 1 else '')
        )
```

The `go_to_page(page)` function:
```python
def go_to_page(page):
    search_state.current_page = page
    # Re-render current view (filtered or unfiltered)
    results = get_current_results()  # Use filtered if active, else full
    start = page * search_state.page_size
    end = start + search_state.page_size
    render_results(results[start:end])
    render_pagination_controls(len(results))
    # Update count display
    results_count.text = f"{tr('Page')} {page + 1}: {start + 1}-{min(end, len(results))} / {len(results)} {tr('Results')}"
```
