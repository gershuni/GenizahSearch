"""PostHog analytics integration for GenizahSearch web app.

- Client-side JS snippet injection for automatic page views + session recordings
- Server-side event helpers for search timing and feature usage

  POSTHOG_API_KEY - Project API key from PostHog dashboard
  POSTHOG_HOST - PostHog instance URL (default: https://app.posthog.com)

When POSTHOG_API_KEY is not set, all analytics calls are no-ops.

import os

POSTHOG_API_KEY = os.environ.get('POSTHOG_API_KEY', '')
POSTHOG_HOST = os.environ.get('POSTHOG_HOST', 'https://app.posthog.com')

def is_enabled() -> bool:
    """Check if PostHog analytics is configured."""
    return bool(POSTHOG_API_KEY)

def track_search_timing_js() -> str:
    """Return JS code snippet to track search execution timing.

    Embed this in the search page to measure perceived search speed.
    Returns empty string if PostHog is disabled.
    """
    if not is_enabled():
        return ""
    return """
    <script>
      window._gsSearchStart = null;
      window.gsTrackSearchStart = function() {
        window._gsSearchStart = performance.now();
      };
      window.gsTrackSearchEnd = function(resultCount) {
        if (window._gsSearchStart && window.posthog) {
          var duration = performance.now() - window._gsSearchStart;
          posthog.capture('search_completed', {
            duration_ms: Math.round(duration),
            result_count: resultCount
          });
          window._gsSearchStart = null;
        }
      };
    </script>
    """

def track_feature_usage_js() -> str:
    """Return JS snippet for tracking feature usage events.

    Call posthog.capture() from NiceGUI event handlers via ui.run_javascript().
    Returns empty string if PostHog is disabled.
    """
    if not is_enabled():
        return ""
    return """
    <script>
      window.gsTrackFeature = function(featureName, properties) {
        if (window.posthog) {
          posthog.capture('feature_used', Object.assign({feature: featureName}, properties || {}));
        }
      };
    </script>
    """
