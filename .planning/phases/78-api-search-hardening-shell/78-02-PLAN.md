**RateLimiter class (D-01 sliding window — NOT the fixed-window puzzle pattern):**
```python
class RateLimiter:
    """Per-IP sliding 60s window using collections.deque[float].

    Process-local state. Production is single uvicorn worker (per CLAUDE.md).
    Multi-worker would require Redis-backed state — out of scope for v7.10.
    """
    def __init__(self, default_limit: int = 30):
        self._buckets: dict[str, deque[float]] = {}
        self._lock = threading.Lock()
        self._default_limit = default_limit

    def _current_limit(self) -> int:
        # Re-read env every call so production can flip without restart (D-02).
        try:
            v = int(os.environ.get('SEARCH_API_RATE_LIMIT', str(self._default_limit)))
            return max(1, v)
        except ValueError:
            return self._default_limit

    def check(self, client_ip: str) -> tuple[bool, int]:
        """Return (allowed, retry_after). retry_after is meaningful only when allowed=False.

        retry_after = max(1, ceil(60 - (now - oldest_in_window))). Always >= 1 second
        when not allowed, so clients can implement an honest backoff per D-01.
        """
        limit = self._current_limit()
        now = time.time()
        cutoff = now - 60.0
        with self._lock:
            dq = self._buckets.get(client_ip)
            if dq is None:
                dq = deque()
                self._buckets[client_ip] = dq
            # Drop entries older than 60s.
            while dq and dq[0] < cutoff:
                dq.popleft()
            if len(dq) >= limit:
                # NOT allowed. Compute Retry-After from the OLDEST entry still in the window.
                oldest = dq[0]
                retry_after = max(1, math.ceil(60.0 - (now - oldest)))
                return (False, retry_after)
            dq.append(now)
            return (True, 0)
```

**Loopback detection (D-03 — loopback ONLY, NO RFC1918):**
```python
def _is_loopback_request(request: Request) -> bool:
    """Return True iff request originates from the local box (127.0.0.1 or ::1).

    D-03: under SEARCH_API_MODE=localhost-only,
        - if request.client.host is loopback, ALSO check XFF first hop is loopback or empty
          (immediate peer is nginx, the only proxy in front).
        - if request.client.host is NOT loopback, ignore XFF and reject.
    """
    if not request.client:
        return False
    direct = (request.client.host or '').strip()
    if direct not in LOOPBACK_IPS:
        return False  # Non-loopback peer: reject regardless of XFF claims.
    xff = (request.headers.get('x-forwarded-for', '') or '').split(',')[0].strip()
    if not xff:
        return True  # Direct loopback connection, no proxy.
    return xff in LOOPBACK_IPS
```

        errs = exc.errors() or [{}]

        message = str(first.get('msg', 'invalid request'))

        return JSONResponse(status_code=400, content=body)

    target_app.add_exception_handler(RequestValidationError, _validation_error_handler)

**Persistent salt (D-11) — mirrors `web/puzzle_tokens.py:18` but adds file persistence:**

```python
def _resolve_posthog_ip_salt() -> str:
    """Read POSTHOG_IP_SALT from env. If absent, auto-generate AND persist to a file
    so hashes survive server restarts (a fresh salt each boot would inflate
    PostHog distinct-user counts — see CLAUDE.md note).
    """
    env_salt = os.environ.get('POSTHOG_IP_SALT', '').strip()
    if env_salt:
        return env_salt
    # No env var: read from / write to disk.
    salt_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), '_secrets')
    salt_path = os.path.join(salt_dir, 'posthog_ip_salt')
    try:
        if os.path.exists(salt_path):
            with open(salt_path, 'r', encoding='utf-8') as f:
                v = f.read().strip()
                if v:
                    return v
        os.makedirs(salt_dir, exist_ok=True)
        new_salt = secrets.token_hex(32)
        # Atomic-ish write: tmp + rename.
        tmp_path = salt_path + '.tmp'
        with open(tmp_path, 'w', encoding='utf-8') as f:
            f.write(new_salt)
        os.replace(tmp_path, salt_path)
        try:
            os.chmod(salt_path, 0o600)  # Best-effort; on Windows this no-ops.
        except OSError:
            pass
        logger.warning(
            "POSTHOG_IP_SALT not set; auto-generated and persisted to %s. "
            "Production should set POSTHOG_IP_SALT explicitly.", salt_path,
        )
        return new_salt
    except OSError as exc:
        # Filesystem write failed (read-only FS, permissions). Fall back to a
        # process-local random salt — restart will rotate it (acceptable for
        # dev / containerized stateless deploys).
        logger.warning("POSTHOG_IP_SALT auto-persist failed (%s); using process-local salt.", exc)
        return secrets.token_hex(32)


_POSTHOG_IP_SALT = _resolve_posthog_ip_salt()


def hash_ip(ip: str) -> str:
    """16-hex-char HMAC-SHA256 of an IP using the persistent salt (D-11)."""
    if not ip:
        ip = 'unknown'
    return hmac.new(_POSTHOG_IP_SALT.encode('utf-8'), ip.encode('utf-8'), hashlib.sha256).hexdigest()[:16]
```

**PostHog event capture (D-10, D-13, D-14):**

```python
_event_queue: queue.Queue = queue.Queue(maxsize=10000)
_drain_thread_started = threading.Event()
_request_counter = 0
_request_counter_lock = threading.Lock()


def _drain_posthog_queue() -> None:
    """Daemon thread: drain _event_queue, POST each event to PostHog. Failures swallowed."""
    api_key = os.environ.get('POSTHOG_API_KEY', '').strip()
    while True:
        try:
            event = _event_queue.get(timeout=60)
        except queue.Empty:
            continue
        if not api_key:
            # No key configured — drain and discard so memory doesn't grow.
            continue
        try:
            payload = {
                'api_key': api_key,
                'event': event['event'],
                'distinct_id': event['distinct_id'],
                'properties': event['properties'],
                'timestamp': event['timestamp'],
            }
            requests.post(POSTHOG_CAPTURE_URL, json=payload, timeout=2.0)
        except Exception:
            # Best-effort observability — never crash the drainer.
            pass


def _start_drain_thread_once() -> None:
    if _drain_thread_started.is_set():
        return
    _drain_thread_started.set()
    t = threading.Thread(target=_drain_posthog_queue, name='posthog-api-drain', daemon=True)
    t.start()


def _should_sample() -> bool:
    """Atomic-ish counter mod N. Default N=1 → every request. D-13."""
    global _request_counter
    try:
        n = max(1, int(os.environ.get('SEARCH_API_POSTHOG_SAMPLE_N', '1')))
    except ValueError:
        n = 1
    with _request_counter_lock:
        _request_counter += 1
        idx = _request_counter
    return (idx % n) == 0


def capture_api_event(
    *,
    endpoint: str,
    mode: Optional[str],
    latency_seconds: float,
    result_count: Optional[int],
    status_code: int,
    error_code: Optional[str],
    client_ip: str,
) -> None:
    """Enqueue a search_api_request PostHog event. Never blocks; never raises.

    Properties (D-14):
      endpoint: 'search' | 'browse' | 'parallels'
      mode: validated mode string or None
      latency_bucket: D-12
      result_count_bucket: D-12 (only if result_count is not None)
      status_code: HTTP status
      error_code: envelope's error.code on non-2xx, else None
    NEVER logged: query, filters, gap, response items, snippets, full text. (HARDEN-05.)
    """
    _start_drain_thread_once()
    if not _should_sample():
        return
    try:
        props: dict = {
            'endpoint': endpoint,
            'mode': mode,
            'latency_bucket': latency_bucket(latency_seconds),
            'status_code': status_code,
            'error_code': error_code,
        }
        if result_count is not None:
            props['result_count_bucket'] = result_count_bucket(result_count)
        event = {
            'event': 'search_api_request',
            'distinct_id': hash_ip(client_ip),
            'properties': props,
            'timestamp': datetime.now(timezone.utc).isoformat(),
        }
        try:
            _event_queue.put_nowait(event)
        except queue.Full:
            # Drop on overflow. PostHog observability is best-effort.
            pass
    except Exception:
        # Never let a logging path crash the request thread.
        pass
```
