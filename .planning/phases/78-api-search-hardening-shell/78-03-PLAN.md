def validate_filter_values(self, filters: dict) -> None:
    if not filters:
        return
    # Lazy import to avoid module-import-time circular dependency (web depends on shared, not vice versa).
    from web.api_hardening import APIError

    domains = filters.get('domains')
    if domains:
        # Canonical key from get_all_domains(): "domain" (verified — see shape note above).
        valid_domains = {d["domain"] for d in (self.get_all_domains() or []) if d.get("domain")}
        for v in domains:
            if v not in valid_domains:
                raise APIError('unresolvable_filter_value',
                               f"filter domains={v!r} not found in FJMS pipeline")

    authors = filters.get('authors')
    if authors:
        # Canonical keys from get_browse_authors(): "person_id" (int) AND "eng_desc" (string).
        # API accepts EITHER form because get_filter_sys_ids dispatches via _is_int(a):
        # digit-string -> person_id FK lookup; otherwise -> AuthorText string equality.
        valid_person_ids: set = set()
        valid_eng_descs: set = set()
        for a in (self.get_browse_authors() or []):
            pid = a.get("person_id")
            if pid is not None:
                valid_person_ids.add(str(pid))  # str-normalize so '17' and 17 both match
            ed = a.get("eng_desc")
            if ed:
                valid_eng_descs.add(ed)
        for v in authors:
            v_str = str(v)
            if v_str.isdigit():
                if v_str not in valid_person_ids:
                    raise APIError('unresolvable_filter_value',
                                   f"filter authors={v!r} not found in FJMS pipeline")
            else:
                if v_str not in valid_eng_descs:

```python
def _discover_valid_materials(self) -> set:
    """Return the set of distinct FragmentMaterial values present in catalog_fields.

    Cached after first call (materials enum is static). Returns empty set if
    self._conn is None — in that case validation degrades to allow-all rather
    than reject-all (the FJMS pipeline itself will simply yield no matches).
    """
    if hasattr(self, '_materials_cache') and self._materials_cache is not None:
        return self._materials_cache
    if self._conn is None:
        self._materials_cache = set()
        return self._materials_cache
    try:
        cursor = self._conn.execute(
            "SELECT DISTINCT FieldValue FROM catalog_fields "
            "WHERE FieldCategory = 'FragmentMaterial' AND FieldValue IS NOT NULL"
        )
        materials = {row[0] for row in cursor if row[0]}
        self._materials_cache = materials
        return materials
    except Exception as e:
        logger.error(f"FjmsService._discover_valid_materials error: {e}")
        self._materials_cache = set()
        return self._materials_cache

def init_search_api(app_override: Optional[FastAPI] = None) -> None:
    """Register Phase 78 search-helper routes onto target_app.

    Args:
        app_override: When None (default, production), registers onto the
                      module-level NiceGUI singleton ``app``. When a bare
                      FastAPI app is passed (test fixtures), registers onto
                      that instead — keeps the NiceGUI singleton clean across
                      multiple TestClient setups (mirroring HIGH-08 from Plan 77-04).
    """
    target_app = app_override if app_override is not None else app

    # Wire the global error envelope onto target_app (D-06).
    register_exception_handlers(target_app)

    @target_app.post('/api/search')
    async def search_endpoint(req: SearchRequest, request: Request):
        t0 = time.monotonic()
        endpoint_name = 'search'
        client_ip = get_client_ip(request)
        # WARNING-4 fix: status_code/error_code are SET in the except branch
        # before the raise propagates. The finally block reads these locals
        # directly — no sys.exc_info() inspection needed.
        status_code = 200
        error_code: Optional[str] = None
        result_count: Optional[int] = None
        validated_mode: Optional[str] = req.mode  # captured for capture_api_event

        try:
            # 1. Mode gate (D-02 re-read every request, D-03 loopback, D-04 disabled→503).
            enforce_mode_gate(request)

            # 2. Rate limit check (D-01 sliding window).
            allowed, retry_after = _rate_limiter.check(client_ip)
            if not allowed:
                raise APIError('rate_limited',
                               'Rate limit exceeded; retry after the specified period.',
                               http_status=429,
                               headers={'Retry-After': str(retry_after)})

            # 3. Query post-validation (D-08, D-09).
            query = (req.query or '').strip()
            if not query:
                raise APIError('query_required', 'query is required and cannot be empty', http_status=400)
            if len(query) > QUERY_LENGTH_CAP:
                raise APIError('query_too_long',
                               f'query exceeds cap (max {QUERY_LENGTH_CAP} chars; submitted {len(query)})',
                               http_status=400)

            if req.limit <= 0:
                raise APIError('invalid_request',
                               f'limit must be positive (submitted {req.limit})',
                               http_status=400)
            if req.limit > MAX_LIMIT:
                raise APIError('limit_too_high',
                               f'limit exceeds max (max {MAX_LIMIT}; submitted {req.limit})',
                               http_status=400)

            # 4. Filter resolution (API-07, D-17).
            #    BLOCKER-3 fix: ONE canonical empty-set short-circuit. If
            #    fjms.get_filter_sys_ids returns an empty set (filters resolvable
            #    but their AND yields zero manuscripts), return count=0 / total=0
            #    / results=[] WITHOUT calling state.searcher.execute_search at all.
            #    Empty intersection is NOT an error per D-17 — it is a valid
            #    response of zero. SearchEngine may treat empty restrict_sys_ids
            #    as "no restriction" (returning all matches), so we MUST NOT pass
            #    an empty set through.
            restrict_sys_ids: Optional[set] = None
            filters_dict: Optional[dict] = None
            short_circuit_empty = False  # set True only when filters yielded empty intersection

            if req.filters is not None:
                # Pydantic v2: model_dump() drops Nones via exclude_none for cleaner FJMS calls.
                filters_dict = req.filters.model_dump(exclude_none=True)
                # Strict reject (D-17). Lazy import to avoid web→shared circular at module-import time.
                from shared.fjms_service import (
                    validate_filter_values,
                    get_fjms_service,
                )
                validate_filter_values(filters_dict)

                # Build restrict_sys_ids using the existing FJMS pipeline (API-07).
                fjms = get_fjms_service(thread_safe=True)
                restrict_sys_ids = fjms.get_filter_sys_ids(
                    domains=filters_dict.get('domains'),
                    authors=filters_dict.get('authors'),
                    works=filters_dict.get('works'),
                    material_include=filters_dict.get('materials'),
                    date_from=filters_dict.get('date_from'),
                    date_to=filters_dict.get('date_to'),
                )
                # FJMS returns None when no filter is active → no restriction (rare here since
                # we entered this branch because req.filters is not None; defensive).
                # FJMS returns set() when filters were resolvable but their AND is empty →
                # ONE canonical short-circuit, no further branches.
                if restrict_sys_ids is not None and len(restrict_sys_ids) == 0:
                    short_circuit_empty = True

            # 5. Build responsa_options if mode='Responsa' (mirroring web/pages/search.py:3920-3930).
            responsa_options = None
            if req.mode == 'Responsa':
                responsa_options = {
                    'responsa_mode': True,
                    'variants': True,
                    'ja': True,
                    'flex_spacing': False,
                    'bidirectional': False,
                    'variant_mode': 'variants',
                }

            # 6. Statelessness check (D-20). Forbidden reads — none of these names appear below.
            #    Asserted by tests/test_search_api.py::test_identical_requests_byte_identical_modulo_timestamp.

            # 7. Execute search (or empty-intersection short-circuit per BLOCKER-3 fix).
            #    The empty-intersection branch MUST NOT call state.searcher.execute_search —
            #    proven by tests/test_search_api.py::
            #    test_filter_resolution_yields_empty_intersection_returns_empty_results_without_executing_search
            #    (asserts execute_search.call_count == 0 in this branch).
            if short_circuit_empty:
                results = []
                total = 0
            else:
                results = state.searcher.execute_search(
                    query_str=query,
                    mode=req.mode,
                    gap=req.gap,
                    progress_callback=None,
                    exclude_words=None,
                    responsa_options=responsa_options,
                    restrict_sys_ids=restrict_sys_ids,
                    text_position=None,
                ) or []
                total = len(results)

            # 8. Cap results to req.limit (HARDEN-02). Total is BEFORE truncation so clients
            #    can see whether more were available.
            results = results[:req.limit]
            result_count = len(results)

            # 9. Lift cascade-downgrade warning from results[0] to top-level (HARDEN-03).
            #    genizah_core.py:7611-7616 currently writes responsa_warning on results[0];
            #    Plan 78 hoists it to the envelope's `warnings:[]` and STRIPS the per-item key
            #    so result rows are clean.
            warnings_list: list = []
            if results:
                first = results[0]
                rw = first.pop('responsa_warning', None)
                if rw:
                    warnings_list.append(f'query_downgraded: {rw}')
                # responsa_expanded_count is also attached to results[0]; not part of the
                # warnings array. Strip it from the API response since it's not in the
                # documented item shape.
                first.pop('responsa_expanded_count', None)

            # 10. Serialize (D-24). The serializer owns the entire envelope shape including
            #     the locator on every item, schema_version, generated_at, etc.
            from shared.search_serializer import serialize_search_payload
            envelope = serialize_search_payload(
                results,
                meta_mgr=state.meta_mgr,
                query=query,
                mode=req.mode,
                gap=req.gap if req.gap else None,
                filters=filters_dict,
                warnings=warnings_list,
                total=total,
            )
            return envelope

        except APIError as exc:
            # WARNING-4 fix: capture status_code + error_code into local variables BEFORE re-raising.
            # The finally block reads these locals — no sys.exc_info() needed, no fragile
            # cross-frame exception inspection. The exception handler in Plan 02 still
            # owns rendering the JSON envelope and emitting Retry-After headers.
            status_code = exc.http_status
            error_code = exc.code
            raise
        except Exception:
            # Catch-all (D-07 internal_error). Log full traceback server-side; emit generic message.
            logger.exception('search_endpoint unhandled exception')
            status_code = 500
            error_code = 'internal_error'
            raise APIError('internal_error', 'Internal error', http_status=500)
        finally:
            # 11. PostHog event capture (HARDEN-05). Exactly once per request, success or error.
            #     Status code + error code are already captured into locals by the except APIError
            #     branch (WARNING-4 fix). No sys.exc_info() inspection needed.
            try:
                elapsed = time.monotonic() - t0
                capture_api_event(
                    endpoint=endpoint_name,
                    mode=validated_mode,
                    latency_seconds=elapsed,
                    result_count=result_count,
                    status_code=status_code,
                    error_code=error_code,
                    client_ip=client_ip,
                )
            except Exception:
                logger.warning('capture_api_event failed in finally block', exc_info=False)

    logger.info("Search API routes initialized: POST /api/search")
```
