```python
def _split_pgp_languages(value: Any) -> list[str]:
    """Split a pgp.db languages_primary/secondary TEXT value into a real list.

    pgp.db stores these as comma-(optionally-)space-separated TEXT, not JSON.
    Sample values seen 2026-05-15:
      None, '', 'Judaeo-Arabic', 'Aramaic, Hebrew', 'Hebrew,Aramaic'
    The existing _build_pgp_subset call site does `list(value or [])` which
    iterates characters for non-empty strings (latent bug). This helper is
    defensive: accepts None / str / list and returns a list[str] of trimmed
    non-empty entries.
    """
    if value is None:
        return []
    if isinstance(value, list):
        return [str(v).strip() for v in value if v and str(v).strip()]
    if isinstance(value, str):
        if not value.strip():
            return []
        return [part.strip() for part in value.split(',') if part.strip()]
    # Unknown type — coerce to single-element list of its str form.
    s = str(value).strip()
    return [s] if s else []


def _pgp_subset_for_sys_id(
    sys_id: str,
    *,
    available_sys_ids: Optional[set[str]] = None,
) -> Optional[dict]:
    """Phase 999.3: shared PGP-lookup-and-project helper for search-results exports.

    Wraps ``shared.document_service.get_document_for_fragment(sys_id, None)`` and
    returns a 10-key projection mirroring ``_build_pgp_subset``'s shape. Used by
    BOTH ``serialize_search_payload`` (JSON envelope) and
    ``web/export_service.py:export_search_results_excel`` (Excel rows) so the two
    surfaces never drift.

    Per D-02 (999.3-CONTEXT.md): does NOT carry transcription text.
    Per D-09: optional ``available_sys_ids`` short-circuit — skip the SQLite hit
    when sys_id is known not to have PGP (caller's hint).

    Args:
        sys_id: GenizahSearch system identifier.
        available_sys_ids: Optional set of sys_ids known to have PGP (e.g.
            ``search_state.transcription_sys_ids``). When provided and
            ``sys_id not in available_sys_ids``, returns None without a DB hit.
            When None, performs the lookup unconditionally. Phase 999.3 callers
            pass None — wiring through ``set_search_export`` is deferred.

    Returns:
        Dict with these 10 keys (R-07 shape, mirroring ``_build_pgp_subset``):
            description, tags, document_type, languages_primary,
            languages_secondary, doc_date_original, doc_date_standard,
            inferred_date_display, pgpid, pgp_url
        Or None when sys_id is empty, short-circuited, has no PGP record, or
        the lookup raises.

    Note on languages format: pgp.db stores ``languages_primary`` and
    ``languages_secondary`` as comma-separated TEXT (e.g. ``'Aramaic, Hebrew'``),
    NOT JSON. This helper splits them into real lists via
    ``_split_pgp_languages`` — fixing a latent bug in the upstream
    ``_build_pgp_subset`` shape (which does ``list(value or [])`` and iterates
    characters for non-empty strings). The upstream function is intentionally
    NOT modified here to avoid regressing the browse JSON envelope.
    """
    if not sys_id:
        return None
    if available_sys_ids is not None and sys_id not in available_sys_ids:
        return None
    try:
        from shared.document_service import get_document_for_fragment
        doc = get_document_for_fragment(sys_id, None)
    except Exception:
        return None
    if not doc:
        return None
    return {
        'description':           doc.get('description'),
        'tags':                  list(doc.get('tags') or []),
        'document_type':         doc.get('document_type'),
        'languages_primary':     _split_pgp_languages(doc.get('languages_primary')),
        'languages_secondary':   _split_pgp_languages(doc.get('languages_secondary')),
        'doc_date_original':     doc.get('doc_date_original'),
        'doc_date_standard':     doc.get('doc_date_standard'),
        'inferred_date_display': doc.get('inferred_date_display'),
        'pgpid':                 doc.get('pgpid'),
        'pgp_url':               doc.get('pgp_url'),
    }
```
