```python
def update_search_export_enrichment(
    transcription_sys_ids: Optional[Any] = None,
    printed_ids: Optional[Any] = None,
    result_domains: Optional[Dict[str, List[str]]] = None,
) -> None:
    """Patch only the 3 enrichment fields (post-search async enrichment sync).

    Phase 999.3 (METADATA-EXPORT-02) — the post-search enrichment block in
    web/pages/search.py:4480-4500 populates `search_state.transcription_sys_ids`
    / `search_state.printed_ids` / `search_state.result_domains` AFTER the
    initial `set_search_export(...)` call at line ~4204. This helper lets that
    block sync the export payload without re-passing results/query/etc.

    Mirrors the Phase 88 D-11 (isinstance guard) + D-12 (copy-on-update)
    invariants from update_search_export_results / update_search_export_selection.
    Silently no-ops if no payload exists (user navigated away mid-enrichment).

    Accepts set or list for the two id args; normalizes to sorted list for
    JSON-safety.
    """
    payload = safe_user_get(_SEARCH_KEY, None)
    if not isinstance(payload, dict):
        return  # D-11: poisoned-shape or missing payload
    payload = dict(payload)  # D-12: copy-on-update
    payload['transcription_sys_ids'] = sorted(transcription_sys_ids) if transcription_sys_ids else []
    payload['printed_ids'] = sorted(printed_ids) if printed_ids else []
    payload['result_domains'] = dict(result_domains) if result_domains else {}
    safe_user_set(_SEARCH_KEY, payload)
```
