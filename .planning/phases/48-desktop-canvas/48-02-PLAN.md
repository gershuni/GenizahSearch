**Part A: Add PuzzleMetaLoaderThread to gui_threads.py**

Add a new `PuzzleMetaLoaderThread(QThread)` after PuzzleImageLoaderThread. This thread resolves fl_ids for a sys_id by fetching NLI data + IIIF manifest asynchronously, so the UI does not block.

```python
class PuzzleMetaLoaderThread(QThread):
    """Resolve images_nli (with fl_ids) for a sys_id in the background.

    FL IDs are NOT available from CSV cache -- they require a network fetch
    of the NLI MARC record and/or IIIF manifest. This thread wraps that
    async resolution so PuzzleCanvasWindow._on_add_shelfmark doesn't block.
    """
    meta_ready = pyqtSignal(str, str, list)   # (sys_id, shelfmark, images_nli_list)
    meta_failed = pyqtSignal(str, str)        # (sys_id, error_message)

    def __init__(self, meta_mgr, sys_id: str, shelfmark: str = ''):
        super().__init__()
        self.meta_mgr = meta_mgr
        self.sys_id = sys_id
        self.shelfmark = shelfmark

    def run(self):
        try:
            # enrich_metadata fetches NLI MARC + IIIF manifest, populating images_nli with fl_ids
            data = self.meta_mgr.enrich_metadata(self.sys_id)
            images_nli = data.get('images_nli', []) if data else []
            if not images_nli:
                # Fallback: try fetch_nli_data directly
                nli_data = self.meta_mgr.fetch_nli_data(self.sys_id)
                if nli_data and nli_data.get('fl_ids'):
                    # Convert fl_ids list to images_nli format
                    images_nli = [{'fl_id': fid, 'label': f'FL{fid}', 'url': ''} for fid in nli_data['fl_ids']]
            if images_nli:
                self.meta_ready.emit(self.sys_id, self.shelfmark, images_nli)
            else:
                self.meta_failed.emit(self.sys_id, "No images found for this manuscript")
        except Exception as e:
            self.meta_failed.emit(self.sys_id, str(e))
```

1. **`add_fragment(self, sys_id, shelfmark, folio_label, fl_id)`**: Public method called by "Add to Puzzle" entry points.
   - Compute `item_key = (sys_id, folio_label)`.
   - **Dedup check**: If `item_key` in `self._fragment_items`, just select that item and return. If `item_key` in `self._pending_fragments`, return (already loading).
   - Create a PuzzleFragment with the given params, x=self._next_x, y=50.0.
   - Store in pending map: `self._pending_fragments[item_key] = puzzle_frag`.
   - Show a placeholder QGraphicsTextItem ("Loading...") on the canvas at the fragment's position.
   - Start a PuzzleImageLoaderThread with the fl_id and default threshold.
   - Connect `thread.image_ready` to `self._on_image_loaded` (pass `item_key` via lambda or functools.partial so the callback can find the pending fragment).
   - Connect `thread.load_failed` to `self._on_image_failed`.
   - Store thread reference in `self._loader_threads` and start it.
   - **Do NOT increment `_next_x` here** — increment in `_on_image_loaded` based on actual pixmap width.

2. **`_on_add_shelfmark(self)`**: Read `self.shelfmark_input.text()`. Normalize via `normalize_shelfmark()`. Look up in `self.app._shelf_to_sys` to get sys_id. If not found, show status bar message "Shelfmark not found" and return.

   **CRITICAL: FL_ID resolution is async.** `get_meta_for_id(sys_id)` returns only (shelfmark, title) -- NO fl_ids. FL IDs require fetching the NLI MARC record and/or IIIF manifest which is a network call. DO NOT call `meta_mgr.enrich_metadata()` on the main thread -- it will freeze the UI.

   Instead, use PuzzleMetaLoaderThread:
   ```python
   def _on_add_shelfmark(self):
       text = self.shelfmark_input.text().strip()
       if not text:
           return
       norm = normalize_shelfmark(text)
       sys_id = self.app._shelf_to_sys.get(norm)
       if not sys_id:
           self.statusBar().showMessage(tr("Shelfmark not found"), 3000)
           return
       shelfmark, _ = self.app.meta_mgr.get_meta_for_id(sys_id)
       if not shelfmark or shelfmark == "Unknown":
           shelfmark = text

       # Check if we already have folio list cached for this sys_id
       if sys_id in self._folio_lists and self._folio_lists[sys_id]:
           images = self._folio_lists[sys_id]
           first = images[0]
           self.add_fragment(sys_id, shelfmark, first.get('label', '1r'), first.get('fl_id', ''))
       else:
           # Need async fl_id resolution
           self.statusBar().showMessage(tr("Resolving images..."), 5000)
           thread = PuzzleMetaLoaderThread(self.app.meta_mgr, sys_id, shelfmark)
           thread.meta_ready.connect(self._on_meta_resolved)
           thread.meta_failed.connect(self._on_meta_failed)
           self._meta_threads.append(thread)
           thread.start()

       self.shelfmark_input.clear()
   ```

3. **`_on_meta_resolved(self, sys_id, shelfmark, images_nli)`**: Callback from PuzzleMetaLoaderThread.
   - Guard: `if sip.isdeleted(self): return`
   - Cache the folio list: `self._folio_lists[sys_id] = images_nli`
   - Take the first image: `first = images_nli[0]`
   - Call `self.add_fragment(sys_id, shelfmark, first.get('label', '1r'), first.get('fl_id', ''))`
   - Clear status bar message.

4. **`_on_meta_failed(self, sys_id, error)`**: Show error in status bar: `self.statusBar().showMessage(tr("Failed to resolve images: {}").format(error), 5000)`

5. **`_on_image_loaded(self, item_key, fl_id, image_bytes)`**: Note: `item_key` is bound via lambda/partial when connecting the signal.
   - Guard: `if sip.isdeleted(self): return`
   - Pop from pending: `puzzle_frag = self._pending_fragments.pop(item_key, None)`. If None, return (was deleted while loading).
   - Convert bytes to QPixmap: `img = QImage(); img.loadFromData(image_bytes); pixmap = QPixmap.fromImage(img)`

   - Create `PuzzleFragmentItem(puzzle_frag, pixmap)`.
   - Add to scene: `self.canvas_view.scene.addItem(item)`.
   - Store in `self._fragment_items[item_key] = item`.
   - **Compute next placement**: `self._next_x = puzzle_frag.x + pixmap.width() * puzzle_frag.scale + 50` (actual width + 50px margin).

```python
def add_to_puzzle(self, sys_id, shelfmark, folio_label=None, fl_id=None):
    """Add a fragment to the puzzle canvas. Opens puzzle window if needed."""
    if self._puzzle_window is None or sip.isdeleted(self._puzzle_window):
        self._puzzle_window = PuzzleCanvasWindow(self)
    if fl_id:
        self._puzzle_window.add_fragment(sys_id, shelfmark, folio_label or '1r', fl_id)
    else:
        # No fl_id provided -- need async resolution via PuzzleMetaLoaderThread
        # Use the same pattern as _on_add_shelfmark but triggered externally
        if sys_id in self._puzzle_window._folio_lists and self._puzzle_window._folio_lists[sys_id]:
            images = self._puzzle_window._folio_lists[sys_id]
            first = images[0]
            self._puzzle_window.add_fragment(sys_id, shelfmark, first.get('label', '1r'), first.get('fl_id', ''))
        else:
            thread = PuzzleMetaLoaderThread(self.meta_mgr, sys_id, shelfmark)
            thread.meta_ready.connect(self._puzzle_window._on_meta_resolved)
            thread.meta_failed.connect(self._puzzle_window._on_meta_failed)
            self._puzzle_window._meta_threads.append(thread)
            thread.start()
    self._puzzle_window.show()
    self._puzzle_window.raise_()
    self._puzzle_window.activateWindow()
```
