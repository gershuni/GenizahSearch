<objective>
Implement desktop session persistence — the critical feature that prevents users from losing search state (exclusions, results, filters) when the app restarts. A power user lost 5,000 exclusions due to zero persistence.

Purpose: The desktop app currently has no state persistence at all. This plan creates a shared session persistence service and integrates it into genizah_app.py so that all search state survives app restarts and crashes.

Output: shared/session_persistence.py service, genizah_app.py save/restore hooks, auto-restore with banner notification on startup.
</objective>

     - `persistence_enabled: bool` (default True)

   - Takes a flat dict matching the SessionState structure
   - path defaults to Config.SESSION_FILE
   - Writes atomically: write to .tmp file then os.replace() to prevent corruption

   - Logs errors but never raises

4. **`clear_session_state(path: str = None) -> bool`**:
   - Deletes the session file if it exists

def _save_session(self):
    """Save current search state to disk for session persistence."""
    from shared.session_persistence import save_session_state
    try:
        cfg = load_app_config()
        if not cfg.get('session_persistence', True):
            return  # Persistence disabled in settings

        state_dict = {
            'version': 1,
            'saved_at': datetime.now().isoformat(),
            'regular_search': {
                'query': self.query_input.text(),
                'mode_index': self.mode_combo.currentIndex(),
                'gap': int(self.gap_input.text()) if self.gap_input.text().isdigit() else 0,
                'variant_preset': getattr(self, '_current_variant_preset', 70),
                'results': self.last_results[:5000],  # Cap at 5K to keep file reasonable
                'domain_exclusions': sorted(self._domain_exclusions),
                'printed_filter': self._printed_filter_state,
                'printed_ids': sorted(self._printed_sys_ids),
                'excluded_sys_ids': sorted(self.excluded_sys_ids),
                'excluded_shelfmarks': sorted(self.excluded_shelfmarks),
                'results_filters': self.results_filters,
                'filter_sources': self.filter_sources,
                'filter_enabled_sources': sorted(self.filter_enabled_sources),
            },
            'composition_search': {
                'source_text': self.comp_text_area.toPlainText(),
                'title': self.comp_title_input.text(),

def _schedule_session_save(self):
    """Schedule a debounced session save."""
    if not hasattr(self, '_session_save_timer'):
        self._session_save_timer = QTimer(self)
        self._session_save_timer.setSingleShot(True)
        self._session_save_timer.timeout.connect(self._save_session)
    self._session_save_timer.start(500)  # 500ms debounce

**Restore hooks — Add `_restore_session()` method** and call it during initialization:

```python
def _restore_session(self):
    """Restore search state from saved session."""
    from shared.session_persistence import load_session_state
    try:
        cfg = load_app_config()
        if not cfg.get('session_persistence', True):
            return

        state = load_session_state()
        if not state:
            return

        # Restore regular search state
        reg = state.get('regular_search', {})
        if reg.get('query'):
            self.query_input.setText(reg['query'])
        if reg.get('mode_index') is not None:
            self.mode_combo.setCurrentIndex(reg['mode_index'])
        if reg.get('gap'):
            self.gap_input.setText(str(reg['gap']))
        if reg.get('results'):
            self.last_results = reg['results']
            # Display restored results using existing display method
            self.display_results(self.last_results)
        self._domain_exclusions = set(reg.get('domain_exclusions', []))
        self._printed_filter_state = reg.get('printed_filter', 'all')
        self._printed_sys_ids = set(reg.get('printed_ids', []))
        self.excluded_sys_ids = set(reg.get('excluded_sys_ids', []))
        self.excluded_shelfmarks = set(reg.get('excluded_shelfmarks', []))
        self.results_filters = reg.get('results_filters', {})
        self.filter_sources = reg.get('filter_sources', {})
        self.filter_enabled_sources = set(reg.get('filter_enabled_sources', []))

        # Restore composition state
        comp = state.get('composition_search', {})
        if comp.get('source_text'):
            self.comp_text_area.setPlainText(comp['source_text'])
        if comp.get('title'):
            self.comp_title_input.setText(comp['title'])
        if comp.get('chunk_size'):
            self.spin_chunk.setValue(comp['chunk_size'])
        if comp.get('max_freq'):
            self.spin_freq.setValue(comp['max_freq'])
        if comp.get('mode_index') is not None:
            self.comp_mode_combo.setCurrentIndex(comp['mode_index'])
        if comp.get('results'):
            self.comp_raw_items = comp['results']
            self.comp_raw_filtered = comp.get('filtered_results', [])
            self.display_comp_results(self.comp_raw_items, {}, {}, self.comp_raw_filtered, {}, {})
        self._comp_domain_exclusions = set(comp.get('domain_exclusions', []))
        self._comp_printed_filter_state = comp.get('printed_filter', 'all')
        self.comp_sort_mode = comp.get('sort_mode', 'score')
        self.comp_sort_reverse = comp.get('sort_reverse', True)

        # Show "Session restored" notification in statusbar (fades after 5 seconds)
        saved_at = state.get('saved_at', '')
        self.statusBar().showMessage(tr("Session restored") + (f" ({saved_at[:16]})" if saved_at else ""), 5000)

    except Exception as e:
        logger.error("Failed to restore session: %s", e)
```
