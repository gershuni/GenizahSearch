```python
def _load_vs(self):
    """D-01/D-05: fetch all VS look-alikes (<=200) for the anchor, store list[Candidate]."""
    from shared.visual_similarity_service import get_vs_service
    from shared.joins_lab import normalize_candidate
    anchor_sid = self.wb._anchor_sid
    if not anchor_sid:
        self._vs_cands = []
        return
    svc = get_vs_service()  # thread_safe=True default — safe; cheap local SQL
    if not svc.is_available() or not svc.has_suggestions(anchor_sid):
        self._vs_cands = []      # D-08 — no VS data
        try:
            self.status.setText(tr("No visual similarity data for this manuscript"))
        except RuntimeError:
            pass
        return
    raw = svc.get_suggestions(anchor_sid, 200)   # D-05: full set, no cap, no floor
    self._vs_cands = [normalize_candidate(_normalize_vs_row(row)) for row in raw]
    try:
        self.status.setText(tr("Visual look-alikes loaded"))
    except RuntimeError:
        pass
```
