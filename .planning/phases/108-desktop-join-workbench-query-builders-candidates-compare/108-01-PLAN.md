# parse_responsa_query(query_str) at :5714 — the public Responsa parser.
#   "(עץ/אילן)" -> [ResponsaComponent(words=['עץ','אילן'])]   (documented at :5727)
#   "#שלום"  -> ResponsaComponent(grammatical_prefixes=True)   (:5722)
#   "שלום#"  -> ResponsaComponent(grammatical_suffixes=True)   (:5723)
#   "%שלום"  -> ResponsaComponent(plene_defective=True)        (:5725)
#   "שלום*"  -> ResponsaComponent(wildcard='suffix')           (:5719)
#   "*נדר"   -> ResponsaComponent(wildcard='prefix')           (:5720)
#   a BARE "עץ|אילן" -> ONE component (words=['עץ|אילן']) then regex-escaped (:6219/:7593) — the BUG.
# ResponsaComponent.words is the OR-alternatives list (:5704).
#
# ⚠ HOISTED GROUP modifiers (RR-13 — the parser's _parse_* at :6113-6163, code-verified):
#   The parser strips a LEADING '-' (negation, :6117-6119), then leading '%'/'#' (:6125-6130), then a
#   TRAILING '#' (:6135-6137) from the WHOLE token, THEN checks for the OR group "(...)" and `after == '*'`
#   for a wildcard-suffix on the group (:6140-6163). So GROUP-LEVEL hoisted forms ALL parse correctly:
#     "#(שלום/שלומות)" -> ResponsaComponent(words=['שלום','שלומות'], grammatical_prefixes=True)   (:6158)
#     "%(a/b)"         -> plene_defective=True on the group
#     "-(עץ/אילן)"     -> negated=True on the OR group (the '-' is stripped before the OR check)
#     "(a/b)#"         -> grammatical_suffixes=True on the group   (:6153-6159)
#     "(a/b)*"         -> wildcard='suffix' on the group           (:6149-6150)
#   But the parser does NOT strip a leading '*' before the OR check, so "*(a/b)" does NOT parse as a
#   wildcard-prefix on the group, AND per-alternative modifiers INSIDE the group ("(#a/%b)") are RAW
#   words from inner.split('/') — i.e. LITERAL '#'/'%' characters. THIS is why RR-13 hoists PER-ROW
#   modifiers OUTSIDE the group; wildcard-PREFIX is the one form that cannot be hoisted onto a multi-box
#   group (single-box rows keep it as "*word").
