With priority-based deduplication per user decision (PGP > Correction > V0.8 > V0.7):

```python
def _deduplicate(self, results):
    """Deduplicate results: one row per manuscript, highest-priority source wins.

    Priority: PGP > Correction > V0.8 > V0.7 (per user decision).
    For PGP results, dedup key is the sys_id from full_header (not pgp:pgpid).
    """
    SOURCE_PRIORITY = {"PGP": 0, "correction": 1, "V0.8": 2, "V0.7": 3}

    def get_dedup_key(r):
        """Get a dedup key that groups results by manuscript."""
        uid = r.get('uid', '')
        # PGP results use pgp:pgpid as uid — extract sys_id from full_header for dedup
        if uid.startswith('pgp:') or uid.startswith('corr:'):
            header = r.get('raw_header', '')
            # full_header is sys_id for PGP/correction docs
            parsed = self.meta_mgr.parse_header_smart(header)
            return parsed[0] if parsed[0] else uid
        return uid

    def get_priority(r):
        src = r.get('display', {}).get('source', 'V0.7')
        return SOURCE_PRIORITY.get(src, 99)

    best = {}  # dedup_key -> result
    for r in results:
        key = get_dedup_key(r)
        if key not in best or get_priority(r) < get_priority(best[key]):
            best[key] = r

    return list(best.values())
```
