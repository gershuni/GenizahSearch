```python
# --- Fetch and index PGP transcriptions ---
pgp_docs_indexed = 0
pgp_fetch_failed = False
try:
    from shared.supabase_provider import get_client
    client = get_client()

    # 1. Fetch fragment mapping (pgpid -> sys_id/shelfmark)
    if progress_callback:
        progress_callback(-1, -1)  # Signal status change
    LOGGER.info("Fetching PGP fragment mapping...")
    frag_mapping = {}  # pgpid -> [{sys_id, shelfmark}, ...]
    frag_offset = 0
    FRAG_PAGE_SIZE = 1000
    while True:
        frag_resp = client.table('document_fragments').select(
            'document_id, sys_id, shelfmark'
        ).range(frag_offset, frag_offset + FRAG_PAGE_SIZE - 1).execute()
        if not frag_resp.data:
            break
        for frag in frag_resp.data:
            pgpid = frag.get('document_id')
            if pgpid not in frag_mapping:
                frag_mapping[pgpid] = []
            frag_mapping[pgpid].append({
                'sys_id': frag.get('sys_id', ''),
                'shelfmark': frag.get('shelfmark', '')
            })
        frag_offset += FRAG_PAGE_SIZE
        if len(frag_resp.data) < FRAG_PAGE_SIZE:
            break
    LOGGER.info("Loaded %d PGP fragment mappings", len(frag_mapping))

    # 2. Fetch PGP documents with transcriptions (batch)
    LOGGER.info("Fetching PGP transcriptions from Supabase...")
    PGP_PAGE_SIZE = 500
    pgp_offset = 0
    while True:
        pgp_resp = client.table('documents').select(
            'pgpid, transcription, shelfmark_combined'
        ).not_.is_('transcription', 'null').neq(
            'transcription', ''
        ).range(pgp_offset, pgp_offset + PGP_PAGE_SIZE - 1).execute()

        if not pgp_resp.data:
            break

        for doc in pgp_resp.data:
            pgpid = doc.get('pgpid')
            raw_text = doc.get('transcription', '')
            if not raw_text or not raw_text.strip():
                continue

            # Preprocess: strip headers, line numbers, brackets
            cleaned = _preprocess_pgp_transcription(raw_text)
            if not cleaned.strip():
                continue

            # Get sys_id and shelfmark from fragment mapping
            frags = frag_mapping.get(pgpid, [])
            if frags:
                first_frag = frags[0]
                sys_id = first_frag.get('sys_id', '')
                shelfmark = first_frag.get('shelfmark', '') or doc.get('shelfmark_combined', '')
            else:
                sys_id = ''
                shelfmark = doc.get('shelfmark_combined', '')

            # Build full_header for display lookup:
            # Format: sys_id so that meta_mgr.parse_header_smart can extract it
            full_header = f"{sys_id}" if sys_id else f"PGP:{pgpid}"

            writer.add_document(tantivy.Document(
                unique_id=f"pgp:{pgpid}",
                content=str(cleaned),
                source="PGP",
                full_header=str(full_header),
                shelfmark=str(shelfmark),
                scope="page",
                boundaries="",
                content_type="pgp"
            ))
            pgp_docs_indexed += 1

        pgp_offset += PGP_PAGE_SIZE
        if progress_callback:
            progress_callback(pgp_offset, pgp_offset + PGP_PAGE_SIZE)
        if len(pgp_resp.data) < PGP_PAGE_SIZE:
            break

    LOGGER.info("Indexed %d PGP transcriptions", pgp_docs_indexed)

    # 3. Fetch approved user corrections
    LOGGER.info("Fetching approved user corrections...")
    corr_resp = client.table('corrections').select(
        'id, sys_id, shelfmark, corrected_text, page_number'
    ).eq('status', 'approved').execute()

    corrections_indexed = 0
    for corr in (corr_resp.data or []):
        corr_text = corr.get('corrected_text', '')
        if not corr_text or not corr_text.strip():
            continue

        sys_id = corr.get('sys_id', '')
        shelfmark = corr.get('shelfmark', '')
        corr_id = corr.get('id', '')

        writer.add_document(tantivy.Document(
            unique_id=f"corr:{corr_id}",
            content=str(corr_text),
            source="correction",
            full_header=str(sys_id),
            shelfmark=str(shelfmark),
            scope="page",
            boundaries="",
            content_type="correction"
        ))
        corrections_indexed += 1

    LOGGER.info("Indexed %d approved corrections", corrections_indexed)

except ImportError:
    LOGGER.warning("Supabase provider not available — skipping PGP/correction indexing")
    pgp_fetch_failed = True
except Exception as e:
    LOGGER.warning("Failed to fetch PGP/correction data: %s — continuing with HTR-only index", e)
    pgp_fetch_failed = True
```

**Add the preprocessing helper function** (as a module-level function near the top of the Indexer class or as a static method):

```python
def _preprocess_pgp_transcription(text):
    """Clean PGP transcription text for indexing.

    Per user decision:
    - Strip Recto/Verso section headers
    - Strip line numbers (leading digits on lines)
    - Strip bracket chars but keep contents: [word] -> word, (note) -> note
    - Strip nikud (follow existing normalization)
    """
    import re
    lines = text.split('\n')
    cleaned_lines = []
    for line in lines:
        stripped = line.strip()
        # Skip section headers (Recto, Verso, recto, verso, and variants like "Recto margin")
        if re.match(r'^(Recto|Verso|recto|verso)(\s+\w+)?\s*$', stripped, re.IGNORECASE):
            continue
        # Strip leading line numbers (e.g., "1. text" or "12 text" at start of line)
        stripped = re.sub(r'^\d+[\.\):]?\s*', '', stripped)
        # Strip brackets but keep contents
        stripped = stripped.replace('[', '').replace(']', '').replace('(', '').replace(')', '')
        # Strip other scholarly marks: {, }, «, »
        stripped = stripped.replace('{', '').replace('}', '').replace('«', '').replace('»', '')
        if stripped:
            cleaned_lines.append(stripped)
    result = '\n'.join(cleaned_lines)
    # Strip nikud using existing function
    result = strip_nikud(result)
    return result
```
