**Step B — Create the gutter widget (`desktop/widgets/line_number_text_edit.py`):**

```python
"""Line-number gutter for QTextEdit/QTextBrowser (desktop transcription panes).

Per Phase 999.4 D-04: the gutter is a SIBLING QWidget — not part of the text
QTextDocument — so user mouse-drag-selection inside the body widget cannot
capture the line numbers. Ctrl+C from a body selection produces only body text.

Per D-10: a "line" is `text.split('\\n')[N-1]` — matches Responsa L<N>:word
semantics at genizah_core.py:4976-4994. Blank lines DO get a number.

Per D-11: numbering restarts at 1 on every apply_line_numbered_text() call.

Persistence (D-09): toggle state stored in app config under key
`show_line_numbers` via genizah_core.load_app_config / save_app_config.
Default True (D-07).
"""
from __future__ import annotations

from typing import Optional

from PyQt6.QtCore import QRect, QSize, Qt
from PyQt6.QtGui import QColor, QFont, QPainter, QPalette
from PyQt6.QtWidgets import QWidget, QTextEdit, QTextBrowser, QAbstractScrollArea

from genizah_core import load_app_config, save_app_config


_CONFIG_KEY = "show_line_numbers"
_GUTTER_PADDING_PX = 8
_GUTTER_FONT_SIZE = 10
_GUTTER_COLOR_HEX = "#9ca3af"
_GUTTER_BG_HEX = "#f3f4f6"  # subtle background to delimit from body


def is_line_numbers_enabled() -> bool:
    """Return current toggle state from app config; default True (D-07)."""
    cfg = load_app_config() or {}
    val = cfg.get(_CONFIG_KEY, True)
    return bool(val)


def set_line_numbers_enabled(enabled: bool) -> None:
    """Persist toggle state to app config (D-09)."""
    save_app_config({_CONFIG_KEY: bool(enabled)})


class LineNumberArea(QWidget):
    """Sibling QWidget painted with one line number per source-text line.

    Lives as a child of the body widget's viewport. Repaints when the body
    scrolls (synced via QAbstractScrollArea.verticalScrollBar().valueChanged).
    Selection-isolated from the body by design — Qt's text cursor cannot
    extend out of its QTextDocument into a sibling widget.
    """

    def __init__(self, body_widget):
        super().__init__(body_widget)
        self._body = body_widget
        self._line_count = 0
        # Track first line of every "block" as currently visible to handle
        # very long files; for our use case (single folio, ~50 lines) we can
        # paint all numbers.
        self._font = QFont()
        self._font.setPointSize(_GUTTER_FONT_SIZE)
        self.setFont(self._font)
        # Match body's vertical scroll
        if isinstance(body_widget, QAbstractScrollArea):
            body_widget.verticalScrollBar().valueChanged.connect(self.update)
        # Repaint when the body's viewport changes size
        body_widget.installEventFilter(self)

    def set_line_count(self, n: int) -> None:
        self._line_count = max(0, int(n))
        self.update()

    def gutter_width(self) -> int:
        """Width needed to fit the largest line number + padding."""
        digits = max(2, len(str(max(1, self._line_count))))
        metrics = self.fontMetrics()
        return metrics.horizontalAdvance("9") * digits + _GUTTER_PADDING_PX * 2

    def sizeHint(self) -> QSize:
        return QSize(self.gutter_width(), 0)

    def paintEvent(self, event):
        painter = QPainter(self)
        painter.fillRect(event.rect(), QColor(_GUTTER_BG_HEX))
        painter.setPen(QColor(_GUTTER_COLOR_HEX))
        painter.setFont(self._font)

        if self._line_count == 0:
            return

        body = self._body
        # Use the body's QTextDocument to find Y positions of each line
        doc = body.document()
        # QTextBlock cursor walking
        block = doc.firstBlock()
        line_no = 1
        scroll_y = body.verticalScrollBar().value() if isinstance(body, QAbstractScrollArea) else 0
        viewport_height = body.viewport().height() if hasattr(body, 'viewport') else self.height()
        gutter_w = self.gutter_width()

        while block.isValid() and line_no <= self._line_count:
            layout = block.layout()
            if layout is not None:
                block_rect = doc.documentLayout().blockBoundingRect(block)
                top = int(block_rect.top()) - scroll_y
                # Each QTextBlock may visually wrap; we want one number per
                # SOURCE line (D-10). Source lines are separated by \n, which
                # in HTML render becomes <br>, which produces multiple
                # QTextBlocks. So one block == one source line in our case
                # (provided we don't insert artificial blocks).
                if top + int(block_rect.height()) > 0 and top < viewport_height:
                    painter.drawText(
                        QRect(0, top, gutter_w - _GUTTER_PADDING_PX, int(block_rect.height())),
                        Qt.AlignmentFlag.AlignRight | Qt.AlignmentFlag.AlignTop,
                        str(line_no),
                    )
            block = block.next()
            line_no += 1


def _ensure_gutter(widget) -> LineNumberArea:
    """Attach a LineNumberArea to widget (idempotent)."""
    existing = getattr(widget, "_line_number_area", None)
    if existing is not None:
        return existing
    area = LineNumberArea(widget)
    widget._line_number_area = area
    # Reserve viewport margin on the appropriate side based on RTL
    _reposition_gutter(widget)
    # Reposition on resize
    _install_resize_hook(widget)
    return area


def _reposition_gutter(widget) -> None:
    """Place the gutter on the RTL leading edge (visually right in RTL).

    In Qt, when layoutDirection == RightToLeft, the viewport's "left" margin
    visually appears on the RIGHT. Setting the viewport's right margin in
    LTR achieves the same visual placement. We always reserve the LEFT
    viewport margin and rely on Qt's automatic RTL flip.

    Actually: QAbstractScrollArea.setViewportMargins(left, top, right, bottom)
    is in PHYSICAL coordinates and is NOT flipped by layoutDirection. So we
    must check the layoutDirection ourselves.
    """
    area = widget._line_number_area
    gutter_w = area.gutter_width()
    is_rtl = widget.layoutDirection() == Qt.LayoutDirection.RightToLeft
    if is_rtl:
        # Gutter on the RIGHT visually = right viewport margin
        widget.setViewportMargins(0, 0, gutter_w if area.isVisible() else 0, 0)
        # Position the area widget in the right strip
        cr = widget.contentsRect()
        area.setGeometry(QRect(cr.right() - gutter_w + 1, cr.top(), gutter_w, cr.height()))
    else:
        # Gutter on the LEFT visually = left viewport margin
        widget.setViewportMargins(gutter_w if area.isVisible() else 0, 0, 0, 0)
        cr = widget.contentsRect()
        area.setGeometry(QRect(cr.left(), cr.top(), gutter_w, cr.height()))


def _install_resize_hook(widget) -> None:
    """Reposition gutter on widget resize."""
    if getattr(widget, "_line_number_resize_hooked", False):
        return
    original_resize = widget.resizeEvent

    def resize_event(event):
        original_resize(event)
        if getattr(widget, "_line_number_area", None) is not None:
            _reposition_gutter(widget)

    widget.resizeEvent = resize_event
    widget._line_number_resize_hooked = True


def apply_line_numbered_text(
    widget,
    rendered_html_or_text: str,
    *,
    source_text: Optional[str] = None,
    is_html: bool = True,
) -> None:
    """Set widget text + render the line-number gutter.

    `widget` — QTextEdit or QTextBrowser
    `rendered_html_or_text` — the body content to display (HTML if is_html=True,
        else plain text). Use the same string the caller would have passed to
        `widget.setHtml(...)` or `widget.setPlainText(...)`.
    `source_text` — the RAW pre-HTML text used for line-counting (D-10).
        If None, `rendered_html_or_text` is used (correct for plain-text mode).
    `is_html` — if True call `setHtml`, else call `setPlainText`.

    Reads toggle state from app config; default True (D-07).
    """
    # Step 1: write the body
    if is_html:
        widget.setHtml(rendered_html_or_text)
    else:
        widget.setPlainText(rendered_html_or_text)

    # Step 2: compute line count from source
    counting_source = source_text if source_text is not None else rendered_html_or_text
    if counting_source:
        # D-10: text.split('\n') — blank lines counted
        line_count = len(counting_source.split("\n"))
    else:
        line_count = 0

    # Step 3: ensure the gutter exists and update count
    area = _ensure_gutter(widget)
    area.set_line_count(line_count)

    # Step 4: visibility from config
    enabled = is_line_numbers_enabled()
    area.setVisible(enabled)
    _reposition_gutter(widget)


def refresh_visibility(widget) -> None:
    """Re-read config and update widget's gutter visibility.

    Call after toggling `show_line_numbers` from a menu/toolbar action.
    """
    area = getattr(widget, "_line_number_area", None)
    if area is None:
        return
    enabled = is_line_numbers_enabled()
    area.setVisible(enabled)
    _reposition_gutter(widget)
```

Notes on common pitfalls:
- `QTextEdit.setHtml(text)` where text contains `<br>` creates QTextBlocks for EACH `<br>`. If our line-count is based on `\n` in `source_text`, and the HTML body has `<br>` separating each source line, the QTextBlock count == source-line count. This is what `_htmlify` does at desktop/result_dialog.py:1840 (`t = text.replace("\n", "<br>")`). For genizah_app.py sites that use `\n`-preserving HTML, the executor MUST verify the block count == source line count, or pass `source_text` explicitly so we don't depend on block parity.
- The paintEvent loop above uses QTextBlock walking — fragile if the HTML inserts non-line blocks (tables, etc). Fallback simpler approach: paint `line_count` numbers evenly spaced by font lineSpacing, ignoring QTextBlock geometry. Trades off perfect alignment for robustness. The executor should pick whichever passes tests AND looks correct in the human-verify step (Task 4).

If QTextBlock alignment proves unreliable, use this simpler fallback in `paintEvent`:

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| transcription text → display | Source comes from local `pgp.db` / FIST.db / corrections — already trusted at storage time; this phase only renders. The existing `_htmlify` in result_dialog.py escapes via `replace`/`re.sub`, not via full HTML escaping — but this is the existing trust boundary, NOT introduced by this phase. |
| toggle state → app config | The boolean `show_line_numbers` is read/written via `load_app_config` / `save_app_config` (pickle). Coerced via `bool(...)` on read. Same trust posture as the existing `show_translations` toggle. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-999.4-06 | Tampering (input) | `apply_line_numbered_text(widget, html, source_text=...)` with malicious source containing control characters | accept | source_text is used ONLY for `len(s.split('\n'))`. No injection surface — count is an integer. The HTML body is passed through unchanged to `setHtml`, which is the existing trust posture (same as direct `setHtml` calls before this phase). No new attack surface. |
| T-999.4-07 | Information Disclosure | gutter QWidget leaks into clipboard via Ctrl+C | mitigate | Sibling-widget design — the LineNumberArea is a child of the body widget but NOT part of its QTextDocument. Qt's text cursor cannot extend out of its document. Verified structurally in Test 3 (`'1\n2\n3' not in qtextedit.toPlainText()`) and via human-verify Ctrl+A+Ctrl+C round-trip. |
| T-999.4-08 | Denial of Service | Pathological transcription (10K+ lines) → 10K painter.drawText calls per scroll | accept | Realistic PGP transcriptions are ≤200 lines per folio. paintEvent draws only line numbers within the viewport rect (via the `if top + height > 0 and top < viewport_height` clip check).
