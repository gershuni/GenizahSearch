```python
def _render_line_numbered_html(
    text: str,
    highlight_html: str | None = None,
    line_height: str = "2.2",
    font_size: str = "1.4rem",
    show_line_numbers: bool = True,
) -> str:
    """Render transcription text with an RTL-aware line-number gutter (D-04, D-10).

    Per D-10 a "line" is `text.split('\\n')[N-1]` — one logical newline.
    Blank lines DO get a number — preserving 1-to-1 alignment with the
    Responsa `L&lt;N&gt;:word` search syntax in genizah_core.py:4976-4994.

    When `show_line_numbers=False`, returns plain rendered HTML with no gutter.

    `highlight_html` (if provided) is the caller's pre-escaped, highlight-marked
    HTML (e.g. from `highlight_text(text)`). When None, `text` is HTML-escaped
    locally. This mirrors the existing browse.py escape contract at :1576-1605.

    The gutter uses CSS grid columns + `user-select: none` on the number span
    so a user drag-selecting inside the body column copies the body only,
    NOT the line numbers (D-04 invariant — testable structurally; see
    tests/test_line_numbers_web.py::test_render_line_numbered_html_copy_paste_invariant).
    """
    import html as html_module
    if not text:
        return ""

    if not show_line_numbers:
        if highlight_html is not None:
            body = highlight_html
        else:
            body = html_module.escape(text).replace("\n", "&lt;br&gt;")
        return (
            f'&lt;div class="transcription-text" style="'
            f'font-size: {font_size}; line-height: {line_height}; '
            f'direction: rtl; text-align: right; '
            f'font-family: \'David\', \'Frank Ruehl\', \'Noto Sans Hebrew\', serif; '
            f'white-space: pre-wrap; overflow-wrap: break-word; word-break: break-word;'
            f'"&gt;{body}&lt;/div&gt;'
        )

    # Line-numbered render
    if highlight_html is not None:
        # Caller pre-built the highlighted HTML; split on the literal &lt;br&gt; or \n
        # boundaries to align with the body. The highlight pipeline at browse.py:1576
        # produces HTML with `\n` preserved (no &lt;br&gt; substitution), so split on `\n`.
        # Defensive: if highlight_html contains &lt;br&gt;, normalize to \n first.
        normalized = highlight_html.replace('&lt;br&gt;', '\n').replace('&lt;br/&gt;', '\n').replace('&lt;br /&gt;', '\n')
        body_lines = normalized.split('\n')
    else:
        escaped = html_module.escape(text)
        body_lines = escaped.split('\n')

    # Gutter numbers — one per body line (D-10: blank lines counted)
    numbers = '\n'.join(str(i + 1) for i in range(len(body_lines)))
    # Body — rejoin with \n (NOT &lt;br&gt;) so white-space: pre-wrap renders them
    body = '\n'.join(body_lines)

    return (
        f'&lt;div class="line-numbered-text" style="'
        f'direction: rtl; text-align: right; '
        f'line-height: {line_height}; font-size: {font_size}; '
        f'font-family: \'David\', \'Frank Ruehl\', \'Noto Sans Hebrew\', serif; '
        f'display: grid; grid-template-columns: max-content 1fr; gap: 0.6em;'
        f'"&gt;'
        f'&lt;span class="line-number-gutter" style="'
        f'color: var(--text-muted, #9ca3af); '
        f'font-size: 0.75em; '
        f'text-align: left; '
        f'font-family: \'Inter\', system-ui, sans-serif; '
        f'user-select: none; -webkit-user-select: none; '
        f'direction: ltr; '
        f'padding-left: 0.4em; '
        f'border-left: 1px solid var(--border-light, #e5e7eb); '
        f'white-space: pre;'
        f'"&gt;{numbers}&lt;/span&gt;'
        f'&lt;div class="line-numbered-body" style="'
        f'white-space: pre-wrap; overflow-wrap: break-word; word-break: break-word;'
        f'"&gt;{body}&lt;/div&gt;'
        f'&lt;/div&gt;'
    )
```

def render_text_section(html_to_render: str):
    """Render pre-formatted HTML text content with optional line-number gutter."""
    from web.safe_storage import safe_user_get
    from web.pages.browse import _render_line_numbered_html  # reuse helper from Task 1
    nonlocal text_content_container
    if text_content_container is None:
        return
    show_ln = bool(safe_user_get('ui.show_line_numbers', True))  # D-07: default ON
    # Source text for line counting comes from current_display_text['value'];
    # html_to_render is the highlight-marked HTML body. Per D-10 the line count
    # is taken from the SOURCE text (split on \n), not the rendered HTML.
    source_text = current_display_text.get('value') or ''
    text_content_container.clear()
    with text_content_container:
        with ui.scroll_area().classes('w-full').style('max-height: 70vh;'):
            if source_text or html_to_render:
                html_str = _render_line_numbered_html(
                    text=source_text,
                    highlight_html=html_to_render or '',
                    line_height="2.4",
                    font_size="1.2rem",
                    show_line_numbers=show_ln,
                )
                # Wrap in the same outer padding the original render used
                with ui.element('div').classes('p-6'):
                    ui.html(html_str, sanitize=False)
            else:
                ui.html('', sanitize=False)
    text_content_container.update()
