    def _on_meas_blur(attr_name, inp_widget, is_int=False):
        def handler(e=None):
            val = inp_widget.value
            if val is not None and val != '':
                val = int(val) if is_int else float(val)
            else:
                val = None
            setattr(search_state, f'filter_{attr_name}', val)
            # Validate min > max (D-19): swap if needed
            _validate_range(attr_name)
            persist_value(f'search_filter_{attr_name}', val)
            asyncio.ensure_future(recompute_fn())
            update_chip_fn()
        return handler

    def _validate_range(attr_name):
        """Normalize min > max per D-19."""
        base = attr_name.rsplit('_', 1)[0]  # e.g., 'width' from 'width_min'
        min_val = getattr(search_state, f'filter_{base}_min', None)
        max_val = getattr(search_state, f'filter_{base}_max', None)
        if min_val is not None and max_val is not None and min_val > max_val:
            setattr(search_state, f'filter_{base}_min', max_val)
            setattr(search_state, f'filter_{base}_max', min_val)

    # Wire blur events (debounced -- only fires when user leaves field)
    width_min_inp.on('blur', _on_meas_blur('width_min', width_min_inp))
    width_max_inp.on('blur', _on_meas_blur('width_max', width_max_inp))
    height_min_inp.on('blur', _on_meas_blur('height_min', height_min_inp))
    height_max_inp.on('blur', _on_meas_blur('height_max', height_max_inp))
