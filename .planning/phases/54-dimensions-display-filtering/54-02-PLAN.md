```python
def get_measurements(self, sys_id: str) -> dict:
    """Returns dict with keys: catalog_sizes, computed, extra_info, blank_images, summary.
    - catalog_sizes: list of dicts with SizeX_cm, SizeY_cm, InnerSizeX_cm, InnerSizeY_cm,

    - computed: list of dicts with FGP, Image_Side, Component_Num, Bifolio_Side,
      Page_Width_cm, Page_Height_cm, Num_Lines, margins (Left/Right/Top/Bottom_Margin_cm),
      Written_Width_cm, Written_Height_cm, Avg_Line_Height_Text_mm, Text_Density_per10cm, DpiGrid
    - extra_info: list of dicts with FGP, Shelfmark, Material, Size_Category,
      NumFolio, NumBifolio, Image_Type
    - blank_images: list of dicts with FGP, Fragment_Width_cm, Fragment_Height_cm, IsNotWhole
    - summary: dict with catalog_width_cm, catalog_height_cm, catalog_inner_width_cm,
      catalog_inner_height_cm, catalog_count, min_computed_width_cm, max_computed_width_cm,
      min_computed_height_cm, max_computed_height_cm, avg_num_lines, min_num_lines,
      max_num_lines, avg_text_density, computed_image_count, material, size_category,
      total_image_count, has_blank_images, blank_image_count (or None)
    """

def has_measurements(self, sys_id: str) -> bool:
    """Check if manuscript has any measurement data.
    Checks manuscript_measurements, catalog_sizes, and computed_measurements.
    Returns False gracefully if tables don't exist."""
```

    ```python
    # web/components/measurements_dialog.py
    import html as html_module
    from nicegui import ui, run
    from web.translations import tr, get_language

    async def show_measurements_dialog(sys_id: str, shelfmark: str, fjms_service=None):
        """Show measurements dialog with catalog + computed measurement data.
        Async to avoid blocking UI thread during data fetch."""
        if fjms_service is None:
            from shared.fjms_service import get_fjms_service
            fjms_service = get_fjms_service(thread_safe=True)

        # Fetch data off the event loop
        data = await run.io_bound(fjms_service.get_measurements, sys_id)
        lang = get_language()
        is_heb = lang == 'he'

        dialog = ui.dialog().props('maximized=false full-width')
        with dialog, ui.card().classes('w-full max-w-[900px] max-h-[90vh]').style(
            'overflow: hidden; display: flex; flex-direction: column;'
        ):
            # Header with teal gradient (distinct from catalog indigo and bib purple)
            with ui.row().classes('w-full items-center justify-between p-3 rounded-t').style(
                'background: linear-gradient(135deg, #00695c, #26a69a); color: white;'
            ):
                with ui.row().classes('items-center gap-2'):
                    ui.icon('straighten').classes('text-xl')
                    ui.label(f'{tr("Measurements")} \u2014 {shelfmark}').classes('text-lg font-bold')
                ui.button(icon='close', on_click=dialog.close).props('flat dense round').classes('text-white')

            # Scrollable content area
            with ui.scroll_area().classes('w-full flex-grow p-4'):
                summary = data.get("summary")
                catalog_sizes = data.get("catalog_sizes", [])
                computed = data.get("computed", [])
                extra_info = data.get("extra_info", [])
                blank_images = data.get("blank_images", [])

                if not summary and not catalog_sizes and not computed and not blank_images:
                    ui.label(tr("No measurement data available")).classes('text-gray-500 italic')
                else:
                    # Section 1: Summary (manuscript-level)
                    if summary:
                        _render_summary_section(summary, is_heb)

                    # Section 2: Catalog Sizes (per D-09, show all with source attribution)
                    if catalog_sizes:
                        _render_catalog_sizes_section(catalog_sizes, is_heb)

                    # Section 3: Computed Measurements (per-image, grouped by Image_Side)
                    if computed:
                        _render_computed_section(computed, extra_info, is_heb)

                    # Section 4: Blank Images (fragments without text blocks)
                    if blank_images:
                        _render_blank_images_section(blank_images, is_heb)

        dialog.open()
    ```

    ```python
    class FjmsMeasurementsDialog(QDialog):
        """Dialog showing physical measurements for a manuscript."""

        def __init__(self, data: dict, sys_id: str = '', shelfmark: str = '', parent=None):
            import html as html_module
            super().__init__(parent)
            self.setWindowTitle(f'{tr("Measurements")} \u2014 {shelfmark}' if shelfmark else tr('Measurements'))
            self.setMinimumSize(700, 450)
            self.resize(800, 600)

            layout = QVBoxLayout(self)
            layout.setContentsMargins(10, 10, 10, 10)

            # Header with teal color (dark mode aware)
            palette = QApplication.palette()
            is_dark = palette.color(QPalette.ColorRole.Window).lightness() < 128
            header_color = '#4db6ac' if is_dark else '#00695c'
            text_color = '#e0e0e0' if is_dark else '#333333'
            header = QLabel(f'<h3 style="color: {header_color};">\U0001f4cf {tr("Measurements")} \u2014 {html_module.escape(shelfmark)}</h3>')
            layout.addWidget(header)

            # Content browser
            browser = QTextBrowser()
            browser.setOpenExternalLinks(False)
            layout.addWidget(browser)

            html_content = self._build_html(data, is_dark, text_color)
            browser.setHtml(html_content)
    ```
