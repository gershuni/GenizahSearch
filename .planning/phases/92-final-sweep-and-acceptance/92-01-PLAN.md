```bash

import ast, pathlib, subprocess, datetime, sys

REPO_ROOT = pathlib.Path('.').resolve()
WEB_DIR = REPO_ROOT / 'web'

def find_app_aliases(tree):
    aliases = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.ImportFrom) and node.module == 'nicegui':
            for n in node.names:
                if n.name == 'app':
                    aliases.add(n.asname or n.name)
    return aliases

def scan_file(path):
    try:
        src = path.read_text(encoding='utf-8')
        tree = ast.parse(src)
    except (SyntaxError, UnicodeDecodeError) as e:
        return [(path, 0, f'PARSE-ERROR: {e}')]
    aliases = find_app_aliases(tree)
    if not aliases:
        return []
    violations = []
    seen_nodes = set()
    for node in ast.walk(tree):
        # Subscript: app.storage.user[key]
        if isinstance(node, ast.Subscript):
            v = node.value
            if (isinstance(v, ast.Attribute) and v.attr == 'user'
                and isinstance(v.value, ast.Attribute) and v.value.attr == 'storage'
                and isinstance(v.value.value, ast.Name) and v.value.value.id in aliases):
                if id(v) in seen_nodes:
                    continue
                seen_nodes.add(id(v))
                snippet = ast.get_source_segment(src, node) or '<no segment>'
                violations.append((path, node.lineno, snippet))
                continue
        # Attribute chain ending at .user (.get/.pop/.bare)
        if isinstance(node, ast.Attribute):
            if node.attr == 'user':
                v = node.value
                if (isinstance(v, ast.Attribute) and v.attr == 'storage'
                    and isinstance(v.value, ast.Name) and v.value.id in aliases):
                    # Skip if parent already handled (e.g., .get() Call wrapping this)
                    if id(node) in seen_nodes:
                        continue
                    seen_nodes.add(id(node))
                    snippet = ast.get_source_segment(src, node) or '<no segment>'
                    violations.append((path, node.lineno, snippet))
    return violations

# Walk web/
files = sorted(WEB_DIR.rglob('*.py'))
total_violations = []
for f in files:
    if f.name == 'safe_storage.py':
        # Skip the chokepoint itself -- it is the ONLY allowed surface
        continue
    total_violations.extend(scan_file(f))

# Repo HEAD short hash
try:
    head = subprocess.check_output(['git', 'rev-parse', '--short', 'HEAD'], cwd=REPO_ROOT).decode().strip()
except Exception:
    head = 'unknown'

# Write evidence
out = pathlib.Path('.planning/phases/92-final-sweep-and-acceptance/92-SWEEP-01-AST-SCAN.txt')

lines.append(f'# Scan timestamp (UTC): {datetime.datetime.utcnow().isoformat()}Z')

lines.append(f'# Allowlist BYPASSED: this raw scan ignores .planning/phase87_storage_allowlist.yaml so the result is unfiltered')
```

```bash
def find_app_aliases(tree):
```
