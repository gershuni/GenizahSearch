2. **Bibliography section**: If `page.metadata.get('bibliography')` has entries, display as a collapsible scholarly bibliography:
   ```python
   bib_data = page.metadata.get('bibliography') or []
   if bib_data:
       ui.separator().classes('my-3')
       with ui.row().classes('items-center gap-2 mb-2'):
           h3(tr('Bibliography'), classes='text-xs font-bold', style='color: var(--text-secondary);')
           ui.badge(str(len(bib_data)), color='indigo').props('outline dense').classes('text-xs')

       # Group by publication title for readability
       from collections import defaultdict
       by_title = defaultdict(list)
       for entry in bib_data:
           title_key = entry.get('title_eng') or entry.get('acronym') or 'Unknown'
           by_title[title_key].append(entry)

       # Show first 5 publications ungrouped, rest in expansion
       max_initial = 5
       title_keys = list(by_title.keys())
       show_keys = title_keys[:max_initial]

       def render_bib_group(title_key, entries):
           for entry in entries:
               # Build mini-citation: "Author, Title (Year) pp. X-Y [Discussion] [Full transcription]"
               parts = []
               authors = entry.get('authors', [])
               if authors:
                   author_names = [a.get('eng') or a.get('heb') or '' for a in authors]
                   parts.append(', '.join(n for n in author_names if n))

               title_display = entry.get('title_eng') or entry.get('acronym') or ''
               year = entry.get('year', '')
               if title_display:
                   if year:
                       parts.append(f'{title_display} ({year})')
                   else:
                       parts.append(title_display)

               page_ref = entry.get('mention_page', '')
               if page_ref:
                   parts.append(f'p. {page_ref}')

               citation = ', '.join(p for p in parts if p)

               with ui.row().classes('gap-2 items-baseline'):
                   ui.label(citation).classes('text-sm').style('color: var(--text-primary);')
                   # Badges for mention type and transcription status
                   mention = entry.get('mention_type', '')
                   if mention and mention != 'Mentioned':
                       color = 'green' if mention == 'Discussion' else 'grey'
                       ui.badge(mention, color=color).props('outline dense').classes('text-xs')
                   trans = entry.get('transcription', '')
                   if trans:
                       ui.badge(f'{trans} transcription', color='teal').props('outline dense').classes('text-xs')
                   transl = entry.get('translation', '')
                   if transl:
                       ui.badge(f'{transl} translation', color='blue').props('outline dense').classes('text-xs')

       for key in show_keys:
           render_bib_group(key, by_title[key])

       if len(title_keys) > max_initial:
           remaining_keys = title_keys[max_initial:]
           remaining_count = sum(len(by_title[k]) for k in remaining_keys)
           with ui.expansion(f'{tr("Show all")} {len(bib_data)} {tr("references")}').classes('text-xs'):
               for key in remaining_keys:
                   render_bib_group(key, by_title[key])
   ```
