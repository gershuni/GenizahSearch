# Start metadata enrichment for Oxford Part manuscript (Phase 33 gap closure)

target_sid_for_enrich = self.current_browse_sid

if cached_meta:
    self.on_browse_enriched_loaded(target_sid_for_enrich, cached_meta)
else:
