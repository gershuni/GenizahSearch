```python
class ReadingDeskWorker(QThread):
    """Batch load PGP sources for multiple fragments for the reading desk."""
    finished = pyqtSignal(list)  # list of (sys_id, sources, pgp_doc)
    error = pyqtSignal(str)

    def __init__(self, sys_ids: list, parent=None):
        super().__init__(parent)
        self.sys_ids = sys_ids  # list of sys_id strings

    def run(self):
        try:
            from shared.document_service import (
                get_all_sources_for_fragment,
                get_document_for_fragment,
            )
            results = []
            for sys_id in self.sys_ids:
                try:
                    sources = get_all_sources_for_fragment(sys_id) or []
                    pgp_doc = get_document_for_fragment(sys_id)
                    results.append((sys_id, sources, pgp_doc or {}))
                except Exception as e:
                    print(f"ReadingDeskWorker: error loading {sys_id}: {e}")
                    results.append((sys_id, [], {}))
            self.finished.emit(results)
        except Exception as e:
            self.error.emit(str(e))
```

def _browse_enter_reading_desk(self, fragments_info=None, pgpid=None):
    """Enter reading desk mode with optional initial fragments.

    Args:
        fragments_info: list of dicts [{shelfmark, document_id}] from JoinsDialog or other sources
        pgpid: PGP document ID if opened from PGP joins
    """
    self.browse_reading_desk_active = True
    self.browse_reading_desk_pgpid = pgpid

    # Disable single-page navigation
    self.btn_b_prev.setEnabled(False)
    self.btn_b_next.setEnabled(False)
    self.combo_browse_page.setEnabled(False)
    self.btn_b_all.setEnabled(False)

    if fragments_info:
        # Populate from provided fragments -- load page data for each
        entries = []
        for idx, frag in enumerate(fragments_info):
            frag_sid = frag.get('document_id', '')
            frag_sm = frag.get('shelfmark', '')
            if not frag_sid:
                continue
            pages = self.searcher.get_full_manuscript(frag_sid)
            entries.append(ReadingDeskEntry(
                sys_id=frag_sid,
                shelfmark=frag_sm,
                pages=pages or [],
                sources=[],  # Will be populated by ReadingDeskWorker
                pgp_doc=None,
                sequence_order=idx
            ))
        self.browse_reading_desk_state.entries = entries
    elif self.current_browse_sid:
        # Start with current manuscript
        shelfmark = None
        if self.meta_mgr:
            try:
                shelfmark, _ = self.meta_mgr.get_meta_for_id(self.current_browse_sid)
            except Exception:
                pass
        pages = self.searcher.get_full_manuscript(self.current_browse_sid)
        self.browse_reading_desk_state.entries = [ReadingDeskEntry(
            sys_id=self.current_browse_sid,
            shelfmark=shelfmark or self.current_browse_sid,
            pages=pages or [],
            sources=[],
            pgp_doc=None,
            sequence_order=0
        )]

    # Render immediately with whatever data we have
    self._browse_rd_render()

    # Launch background worker to load PGP sources for all fragments
    sys_ids = [e.sys_id for e in self.browse_reading_desk_state.entries]
    if sys_ids:
        self._browse_rd_worker = ReadingDeskWorker(sys_ids)
        self._browse_rd_worker.finished.connect(self._browse_rd_on_sources_loaded)
        self._browse_rd_worker.error.connect(lambda err: print(f"ReadingDeskWorker error: {err}"))
        self._browse_rd_worker.start()

def _browse_rd_on_sources_loaded(self, results):
    """Callback when ReadingDeskWorker finishes loading PGP sources."""
    if not self.browse_reading_desk_active:
        return
    # Update entries with loaded sources (ReadingDeskEntry dataclass attributes)
    for sys_id, sources, pgp_doc in results:
        for entry in self.browse_reading_desk_state.entries:
            if entry.sys_id == sys_id:
                entry.sources = sources
                entry.pgp_doc = pgp_doc
                break
    # Re-render with sources (version selectors will now show options)
    self._browse_rd_render()

**CODEBASE-VERIFIED IMAGE LOADING APPROACH:**
The `ZoomableScrollArea` (line 1273 of genizah_app.py) is a `QGraphicsView` with these methods:
- `set_image(pixmap: QPixmap)` -- displays an image with fit-to-viewport
- `set_status_message(text: str)` -- shows text overlay (e.g., "Loading...", "No Image")
- `zoom_in()` / `zoom_out()` -- zoom by factor 1.2 / 0.8
- `set_rotation(angle: float)` -- set absolute rotation in degrees
- `rotate_view(degrees)` -- add degrees to current rotation

It does NOT have `load_images`, `load_images_for_page`, or any network loading method. Image loading is handled externally by `ImageLoaderThread` (line 1797), which takes a URL, downloads with caching, and emits `image_loaded(QImage)`.

def _browse_rd_show_version_dialog(self, sys_id):
    """Show a simple dialog to select PGP version for a fragment."""
    entry = next((e for e in self.browse_reading_desk_state.entries if e.sys_id == sys_id), None)
    if not entry or not entry.sources:
        QMessageBox.information(self, tr("Versions"), tr("No alternate versions available"))
        return

    sources = entry.sources
    items = []
    for i, s in enumerate(sources):
        desc = s.get('source_description', '') or s.get('doc_relation', '')
        lang = s.get('language', '')
        label = f"{desc} ({lang})" if lang else desc
        items.append(label)

    if not items:
        return

    from PyQt6.QtWidgets import QInputDialog
    selected, ok = QInputDialog.getItem(
        self, tr("Select Version"), tr("Choose transcription source:"),
        items, 0, False
    )
    if ok and selected:
        idx = items.index(selected)
        source = sources[idx]
        # Update the pages for this entry with the selected source's content
        new_text = source.get('content', '') or source.get('transcription', '')
        if new_text:
            # Replace page text with source text
            if entry.pages:
                # Set all pages to this source's text (PGP sources are typically full-document)
                for p in entry.pages:
                    if hasattr(p, 'text'):
                        p.text = new_text
                    elif isinstance(p, dict):
                        p['text'] = new_text
            self._browse_rd_render()
