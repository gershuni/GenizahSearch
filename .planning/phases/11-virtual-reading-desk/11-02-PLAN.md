# "Add to View" button -- starts or adds to reading desk
def add_to_reading_desk():
    if not state.sys_id:
        return
    current_sm = page.shelfmark or state.sys_id
    current_sid = state.sys_id
    if state.view_joined:
        # Already in reading desk -- add current manuscript if not already present
        existing_ids = {e.get('sys_id') for e in state.reading_desk_entries}
        if current_sid in existing_ids:
            ui.notify(tr('Already in view'), type='info')
            return
        # Load and add
        from shared.document_service import get_all_sources_for_fragment, get_document_for_fragment
        pages = service.get_full_manuscript(current_sid)
        sources = []
        pgp_doc = None
        try:
            sources = get_all_sources_for_fragment(current_sid) or []
            pgp_doc = get_document_for_fragment(current_sid)
        except Exception:
            pass
        state.reading_desk_entries.append({
            'sys_id': current_sid,
            'shelfmark': current_sm,
            'pages': pages or [],
            'sources': sources,
            'pgp_doc': pgp_doc or {}
        })
        update_content()
    else:
        # Start reading desk with current manuscript
        enter_joined_view([{'shelfmark': current_sm, 'document_id': current_sid}])

ui.button(tr('Add to View'), icon='playlist_add', on_click=add_to_reading_desk).props(
    'flat dense'
).classes('text-sm').tooltip(tr('Add current manuscript to Reading Desk'))

2. **"Add from List" dialog:**
   ```python
   def show_add_from_list_dialog():
       """Show dialog to select manuscripts from personal lists."""
       from web.state import state as app_state
       if not app_state.lists_mgr:
           ui.notify(tr('Please log in to access your lists'), type='warning')
           return

       with ui.dialog() as dialog, ui.card().style('min-width: 400px; max-height: 70vh;'):
           ui.label(tr('Add from Personal List')).classes('text-lg font-bold mb-2')

           # Get user's lists
           try:
               lists = app_state.lists_mgr.get_lists_sync()
           except Exception:
               lists = []

           if not lists:
               ui.label(tr('No lists found')).classes('italic text-gray-500')
           else:
               for lst in lists:
                   list_id = lst.get('id')
                   list_name = lst.get('name', tr('Unnamed List'))
                   item_count = lst.get('item_count', 0)

                   def add_list_items(lid=list_id, lname=list_name):
                       items = app_state.lists_mgr.get_items_in_list_sync(lid)
                       if not items:
                           ui.notify(tr('No items in this list'), type='info')
                           return
                       existing_ids = {e.get('sys_id') for e in state.reading_desk_entries}
                       added = 0
                       from shared.document_service import get_all_sources_for_fragment, get_document_for_fragment
                       for item in items:
                           item_sid = item.get('sys_id', '')
                           if not item_sid or item_sid in existing_ids:
                               continue
                           item_sm = item.get('shelfmark') or item_sid
                           pages = service.get_full_manuscript(item_sid)
                           sources = []
                           pgp_doc = None
                           try:
                               sources = get_all_sources_for_fragment(item_sid) or []
                               pgp_doc = get_document_for_fragment(item_sid)
                           except Exception:
                               pass
                           state.reading_desk_entries.append({
                               'sys_id': item_sid,
                               'shelfmark': item_sm,
                               'pages': pages or [],
                               'sources': sources,
                               'pgp_doc': pgp_doc or {}
                           })
                           state.joined_fragments_info.append({'shelfmark': item_sm, 'document_id': item_sid})
                           existing_ids.add(item_sid)
                           added += 1
                       if added:
                           ui.notify(f'{added} {tr("manuscripts added")}', type='positive')
                           dialog.close()
                           update_content()
                       else:
                           ui.notify(tr('All items already in view'), type='info')

                   with ui.row().classes('w-full items-center gap-2 py-1 cursor-pointer hover:bg-gray-100 px-2 rounded'):
                       ui.icon('folder').classes('text-green-600')
                       ui.label(f'{list_name}').classes('flex-grow')
                       ui.badge(str(item_count)).props('outline dense')
                       ui.button(icon='add', on_click=add_list_items).props('flat dense round size=sm color=green')

           with ui.row().classes('w-full justify-end mt-3'):
               ui.button(tr('Close'), on_click=dialog.close).props('flat')

       dialog.open()
   ```

   To preserve reading desk state across language changes:

   - OR intercept the language switch to NOT refresh/navigate when reading desk is active -- instead just update translations in place.
   - The simplest approach: in `create_browse_page`, at the start, check if there's a saved reading desk state in URL params or storage and restore it.

   Inspect the language switching mechanism:

   - If it's via in-page toggle, state should persist.

   Solution: Before language switch, serialize the reading desk sys_ids. After switch/reload, detect and restore. Store as URL query parameter or use app.storage:
