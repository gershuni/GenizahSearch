        assert "/browse" in paths, f"'/browse' not in OpenAPI spec paths: {paths}"

    @pytest.mark.skipif(not _IMPORT_OK, reason="nicegui/fastapi not importable in this env")
    def test_openapi_excludes_legacy_routes():
        """GET /api/openapi.json must NOT contain legacy /api/* routes."""
        client = TestClient(nicegui_app)
        resp = client.get("/api/openapi.json")
        assert resp.status_code == 200
        paths = set(resp.json().get("paths", {}).keys())
        forbidden = {"/api/cambridge_image/{sys_id}", "/robots.txt", "/sitemap.xml",
                     "/_internal/memstat", "/api/nli_image_by_sysid"}

        assert not leaked, f"Legacy routes leaked into OpenAPI spec (D-07 violation): {leaked}"
