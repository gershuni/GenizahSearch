    ```python
    def normalize_join_source(j: dict) -> str:
        s = (j.get("source") or "").strip()
        if s:
            return s
        return "user" if j.get("is_local") else ""

    def build_known_join_rows(user_joins, pgp_joins, fjms_joins, anchor_sid, anchor_shelf, meta_mgr=None):
        """Merge the three join lists (dedup by pair), return one row per unique pair.
        Each row: {fragment_a, fragment_b, source, other_sid, other_shelf}.
        Call order = badge priority: PGP > FJMS > user (first source wins on dedup)."""
        merged = dedup_join_rows([pgp_joins or [], fjms_joins or [], user_joins or []])
        rows = []
        a_norm = (anchor_shelf or "").upper().strip()
        for j in merged:
            fa = j.get("fragment_a") or ""
            fb = j.get("fragment_b") or ""
            # the OTHER member is the one that is not the anchor
            other = fb if fa.upper().strip() == a_norm else fa
            other_sid = j.get("other_sid") or ""
            if not other_sid and meta_mgr is not None:
                try:
                    other_sid = (meta_mgr.resolve_system_by_shelfmark(other) or {}).get("sys_id", "")
                except Exception:
                    other_sid = ""
            rows.append({
                "fragment_a": fa, "fragment_b": fb,
                "source": normalize_join_source(j),
                "other_sid": other_sid, "other_shelf": other,
            })
        return rows
    ```
