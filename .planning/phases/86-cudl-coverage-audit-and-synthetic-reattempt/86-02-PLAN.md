    def _guess_pattern(cudl_classmark: str) -> str:
        """Categorize residue classmark into one of the 5 known D-02b families.

        Output is a hint for human adjudicators looking at 86-RESIDUE-PATTERNS.md
        (Phase 86 Plan 03); not a load-bearing decision. Returns 'other' for
        classmarks that don't match any of the known prefixes.
        """
        if not cudl_classmark:
            return "other"
        if cudl_classmark.startswith("tsf"):
            return "tsf_flattened_series"
        if cudl_classmark.startswith("tsar"):
            return "tsar_flattened_series"
        if cudl_classmark.startswith("tsns"):
            if "minute" in cudl_classmark or cudl_classmark[-1:].isalpha():
                return "tsns_minute_or_letter"
            return "tsns_other"
        if cudl_classmark.startswith("or"):
            return "or_single_segment"
        if cudl_classmark.startswith("mosseri"):
            return "mosseri_exotic_letter"
        if cudl_classmark.startswith("tsmisc"):
            return "tsmisc_multi_segment"
        return "other"
    ```

    def _build_qualifying_inventories(
        fist_conn: sqlite3.Connection,
        nli_conn: Optional[sqlite3.Connection] = None,
    ) -> tuple[dict[int, dict], list[dict]]:
        """CUDL-WALKED rewrite (Phase 86 D-01).

        Walks nli_crossref.db.cambridge_manifests; for each classmark:
          1. lookup_cudl(classmark) -> if hit, SKIP (Phase 84 already in libraries.csv).
          2. lookup_fist_by_cudl(classmark) -> resolve to FIST InventoryId via D-02a normalizers.
             - If None: log residue with pattern_guess column (D-02c).
             - If rec.has_alma: SKIP (libraries.csv row exists — alias-only gap, not a synthetic).
             - If rec.fist_shelfmark in parent_shadow set: SKIP (D-06 filter).
             - Else emit qualifying row keyed by rec.inventory_id (D-01a image-bearing-only).

        Returns: (qualifying: dict[int, dict], residue: list[dict])
        """
        if nli_conn is None:
            raise ValueError(
                "Phase 86 D-01 CUDL-walk requires nli_conn (nli_crossref.db). "
                "Phase 85's FIST-walk fallback is removed in Phase 86."
            )

        # Step 1: Walk CUDL classmarks (deterministic ORDER BY normalized_shelfmark).
        cudl_rows = list(nli_conn.execute(
            "SELECT label, manifest_url, normalized_shelfmark FROM cambridge_manifests "
            "WHERE normalized_shelfmark IS NOT NULL AND normalized_shelfmark != '' "
            "ORDER BY normalized_shelfmark"
        ))

        # Step 2: Build Phase 86 FIST<->CUDL alias index (in-memory, one-shot).
        build_fist_alias_index(fist_conn)

        # Step 3: Load D-06 parent-shadow filter.
        parent_shelfmarks = _load_parent_shelfmark_set()

        qualifying: dict[int, dict] = {}
        residue: list[dict] = []

        for label, manifest_url, classmark in cudl_rows:
            # Phase 84 check — already in libraries.csv.
            if lookup_cudl(classmark) is not None:
                continue

            # Phase 86 FIST resolution.
            rec = lookup_fist_by_cudl(classmark)
            if rec is None:
                residue.append({
                    "inventory_id": "",
                    "signature_id": "",
                    "ambiguity_kind": "no_fist_match",
                    "classmark": classmark,
                    "cudl_label": label,
                    "fist_signature_ids": "",
                    "fist_inventory_ids": "",
                    "leading_char": "",
                    "pattern_guess": _guess_pattern(classmark),
                })
                continue

            # D-01a check: if FIST inventory has Alma -> libraries.csv row exists already.
            if rec.has_alma:
                # NOT a synthetic candidate — Phase 84 alias gap, future work.
                continue

            # D-06 parent-shadow filter.
            if rec.fist_shelfmark in parent_shelfmarks:
                residue.append({
                    "inventory_id": str(rec.inventory_id),
                    "signature_id": "",
                    "ambiguity_kind": "parent_shadow",
                    "classmark": classmark,
                    "cudl_label": label,
                    "fist_signature_ids": "",
                    "fist_inventory_ids": str(rec.inventory_id),
                    "leading_char": "",
                    "pattern_guess": _guess_pattern(classmark),
                })
                continue

            # CSV-injection fail-loud check (preserved from Phase 85).
            if _has_csv_injection_leader(rec.fist_shelfmark) or _has_csv_injection_leader(label):
                residue.append({
                    "inventory_id": str(rec.inventory_id),
                    "signature_id": "",
                    "ambiguity_kind": "csv_injection_leader",
                    "classmark": classmark,
                    "cudl_label": label,
                    "fist_signature_ids": "",
                    "fist_inventory_ids": str(rec.inventory_id),
                    "leading_char": (rec.fist_shelfmark or label or "")[:1],
                    "pattern_guess": _guess_pattern(classmark),
                })
                continue

            qualifying[rec.inventory_id] = {
                "canonical_shelfmark": rec.fist_shelfmark,
                "title_heb": None,
                "title_eng": None,
                "genizah_title": None,
                "library_code": _classify_library_code(rec.fist_shelfmark),
                "has_cudl_manifest": True,         # D-01a invariant
                "has_fjms_metadata": False,        # populated downstream by export_fist_enrichment.py
                "cudl_label": label,
            }

        return qualifying, residue
    ```
