    ```python
    """Phase 86 Plan 03: build the D-02c human-in-the-loop adjudication artifact.

    Reads reports/synthetic_ambiguity_residue.csv (Plan 02 output) and writes
    .planning/phases/86-cudl-coverage-audit-and-synthetic-reattempt/86-RESIDUE-PATTERNS.md
    with per-family fixture tables. Each fixture shows a CUDL classmark + 3
    nearest-neighbour FIST candidates by prefix + numeric-substring matching
    (NOT Levenshtein — per RESEARCH.md alternatives table).

    User adjudicates each family in-place (Accept / Reject / Spot-check); the
    outcome rolls into shared/fist_cudl_bridge.py + reports/cudl_coverage.md.
    """
    from __future__ import annotations
    import csv
    import sqlite3
    import sys
    from collections import defaultdict
    from pathlib import Path

    ROOT = Path(__file__).resolve().parent.parent
    RESIDUE_CSV = ROOT / "reports" / "synthetic_ambiguity_residue.csv"
    OUT_PATH = (
        ROOT / ".planning" / "phases"
        / "86-cudl-coverage-audit-and-synthetic-reattempt"
        / "86-RESIDUE-PATTERNS.md"
    )
    FIST_DB = ROOT / "fist_data" / "FIST.db"

    PATTERN_FAMILIES = [
        ("tsf_flattened_series", "T-S F flattened-series hypothesis",
         "CUDL `tsf1.1100` may correspond to FIST `T-S F1(1).100` or `T-S F1(2).100`"
         " (the leading `1` may encode the FIST `(N)` series digit)."),
        ("tsar_flattened_series", "T-S Ar flattened-series hypothesis",
         "Same shape as T-S F: CUDL flat suffix may encode FIST `(N)` series digit."),
        ("tsns_minute_or_letter", "T-S NS minute-fragments + letter suffixes",
         "FIST writes `T-S NS 192.minute fragments` form (phrase suffix)."
         " Letter-suffix patterns (`tsns135.1aa`, `tsns135.1ab`) may map differently."),
        ("or_single_segment", "Or. single-segment ambiguity",
         "CUDL `or1080.11` vs FIST `Or.1080 11.1` — different fragment granularities?"),
        ("mosseri_exotic_letter", "Mosseri exotic letter suffixes",
         "`mosseriii117.1a`, `mosseriv270b` — FIST may have variant forms not probed."),
        ("tsmisc_multi_segment", "T-S Misc multi-segment patterns",
         "`tsmisc1.131.1`, `tsmisc24.137.21` — likely Or.-like multi-segment normalizer."),
    ]


    def load_residue_by_family() -> dict[str, list[dict]]:
        if not RESIDUE_CSV.exists():
            print(f"FATAL: residue CSV not found at {RESIDUE_CSV}. "
                  f"Run `python scripts/generate_synthetic_rows.py --dry-run` first.",
                  file=sys.stderr)
            sys.exit(2)
        by_family: dict[str, list[dict]] = defaultdict(list)
        with RESIDUE_CSV.open("r", encoding="utf-8", newline="") as f:
            for row in csv.DictReader(f):
                fam = (row.get("pattern_guess") or "other").strip()
                by_family[fam].append(row)
        return by_family


    def nearest_fist_candidates(classmark: str, fist_conn: sqlite3.Connection,
                                limit: int = 3) -> list[str]:
        """Return up to N FIST shelfmarks heuristically near `classmark`.

        Prefix-matching approach (per RESEARCH.md alternatives table); no
        Levenshtein. Uses first 5 letters of the classmark as the LIKE pattern.
        """
        prefix = classmark[:5] if classmark else ""
        if not prefix:
            return []
        # Use case-insensitive LIKE; FIST.dbo_Inventory.Shelfmark stores mixed case.
        rows = fist_conn.execute(
            "SELECT Shelfmark FROM dbo_Inventory "
            "WHERE Shelfmark IS NOT NULL "
            "  AND LOWER(REPLACE(REPLACE(Shelfmark, ' ', ''), '-', '')) LIKE ? "
            "ORDER BY Shelfmark "
            "LIMIT ?",
            (f"%{prefix.lower()}%", limit),
        ).fetchall()
        return [r[0] for r in rows]


    def build_artifact() -> None:
        by_family = load_residue_by_family()
        fist_conn = sqlite3.connect(f"file:{FIST_DB}?mode=ro", uri=True) if FIST_DB.exists() else None

        lines = [
            "# Phase 86 Residue Pattern Adjudication",
            "",
            "**Generated:** auto by `scripts/build_residue_patterns_artifact.py`",
            "**Source residue:** `reports/synthetic_ambiguity_residue.csv`"
            f" (Phase 86 rebuild; {sum(len(v) for v in by_family.values())} entries)",
            "**Adjudication target:** 5 pattern families × Accept/Reject/Spot-check",
            "",
            "Each section below shows up to 5 sample CUDL classmarks from the residue,"
            " each paired with up to 3 nearest-neighbour FIST shelfmarks (prefix-match"
            " heuristic — these are HINTS, not authoritative mappings). The user"
            " adjudicates whether the pattern family represents a recoverable"
            " encoding gap or a legitimate fragment-granularity difference.",
            "",
        ]

        for fam_key, fam_title, fam_desc in PATTERN_FAMILIES:
            entries = by_family.get(fam_key, [])
            lines.append(f"## Pattern Family: {fam_title} ({len(entries)} entries)")
            lines.append("")
            lines.append(f"**Hypothesis:** {fam_desc}")
            lines.append("")
            lines.append("**Sample fixtures (up to 5 classmarks × 3 FIST candidates each):**")
            lines.append("")
            lines.append("| CUDL classmark | FIST candidate 1 | FIST candidate 2 | FIST candidate 3 |")
            lines.append("| -------------- | ---------------- | ---------------- | ---------------- |")
            for row in entries[:5]:
                classmark = row.get("classmark", "")
                cands = (
                    nearest_fist_candidates(classmark, fist_conn, limit=3)
                    if fist_conn else []
                )
                cands = cands + [""] * (3 - len(cands))
                lines.append(
                    f"| `{classmark}` | `{cands[0]}` | `{cands[1]}` | `{cands[2]}` |"
                )
            lines.append("")
            lines.append("**User decision:** [ ] Accept rule  [ ] Reject  [ ] Spot-check more")
            lines.append("")
            lines.append("**Proposed normalizer rule (if Accepted, edit before locking):**")
            lines.append("")
            lines.append("```")
            lines.append("# Fill in: regex / transformation that recovers this family.")
            lines.append("```")
            lines.append("")
            lines.append("**Rejection rationale (if Rejected):**")
            lines.append("")
            lines.append("```")
            lines.append("# Fill in: why this family is genuinely residual, not encoding gap.")
            lines.append("```")
            lines.append("")

        # Footer with operational guidance.
        lines.extend([
            "---",
            "",
            "## After Adjudication",
            "",
            "Accepted rules: Phase 86 Plan 03 Task 2 integrates them into"
            " `shared/fist_cudl_bridge.py::fist_to_cudl_keys` with matching unit tests"
            " in `tests/test_fist_cudl_bridge.py`. Plan 04 then re-runs"
            " `python scripts/generate_synthetic_rows.py --apply`.",
            "",
            "Rejected and Deferred rules: documented in this artifact (preserved) and"
            " referenced in `reports/cudl_coverage.md` (Plan 04) under"
            " 'Residue Pattern Adjudication' so future maintainers know they were"
            " evaluated and excluded by design.",
            "",
        ])

        OUT_PATH.write_text("\n".join(lines) + "\n", encoding="utf-8")
        print(f"Wrote {OUT_PATH}")
        if fist_conn is not None:
            fist_conn.close()


    if __name__ == "__main__":
        build_artifact()
    ```
