    def map_cudl_prefix_to_fist(classmark: str) -> str | None:
        """Return the FIST shelfmark-prefix that CUDL `classmark` should map to,
        or None if no known mapping (the residue artifact omits FIST candidates
        for such classmarks)."""
        cm = (classmark or "").lower()
        for cudl_prefix, fist_prefix in _CUDL_TO_FIST_PREFIX:
            if cm.startswith(cudl_prefix):
                return fist_prefix
        return None


    def numeric_tokens(s: str) -> list[str]:
        return _NUM_TOKEN_RE.findall(s or "")


    def nearest_fist_candidates(
        classmark: str,
        fist_conn: sqlite3.Connection,
        limit: int = 3,
    ) -> list[dict]:
        """BRIDGE-AWARE nearest-neighbour for `classmark`.

        Returns list of {fist_shelfmark, inventory_id, signature_id, title, genizah_title}.
        Strategy (per HIGH #2):
          1. Map CUDL prefix -> FIST prefix; if no mapping, return [].
          2. Extract numeric tokens from classmark.
          3. SELECT dbo_Inventory rows whose Shelfmark starts with FIST prefix
             (case-insensitive) AND contains at least one numeric token.
          4. Rank by numeric-token-overlap score (descending), limit to N.
          5. Join InventorySignature + Signature + UnitCatalogRec for context.
        """
        fist_prefix = map_cudl_prefix_to_fist(classmark)
        if not fist_prefix:
            return []

        tokens = numeric_tokens(classmark)
        if not tokens:
            return []

        # Step 1: pull all candidate FIST shelfmarks starting with the mapped prefix.
        # The prefix LIKE is bounded; FIST has ~280K inventories total but only a
        # small fraction share any given prefix.
        rows = fist_conn.execute(
            "SELECT inv.InventoryId, inv.Shelfmark, sig.SignatureId, "
            "       ucr.Title, ucr.GenizahTitleText "
            "FROM dbo_Inventory inv "
            "LEFT JOIN dbo_InventorySignature isig ON isig.InventoryId = inv.InventoryId "
            "LEFT JOIN dbo_Signature sig ON sig.SetSignatureId = isig.SetSignatureId "
            "LEFT JOIN dbo_UnitCatalogRec ucr ON ucr.SignatureId = sig.SignatureId "
            "WHERE LOWER(inv.Shelfmark) LIKE ? "
            "ORDER BY inv.InventoryId "
            "LIMIT 500",
            (f"{fist_prefix.lower()}%",),
        ).fetchall()

        # Step 2: rank by numeric-token overlap (descending).
        scored: list[tuple[int, dict]] = []
        for inv_id, shelfmark, sig_id, title, gtitle in rows:
            fist_tokens = numeric_tokens(shelfmark)
            overlap = len(set(tokens) & set(fist_tokens))
            if overlap == 0:
                continue
            scored.append((overlap, {
                "fist_shelfmark": shelfmark,
                "inventory_id": inv_id,
                "signature_id": sig_id,
                "title": (title or "")[:60],
                "genizah_title": (gtitle or "")[:60],
            }))
        scored.sort(key=lambda x: (-x[0], x[1]["inventory_id"]))
        return [s[1] for s in scored[:limit]]
