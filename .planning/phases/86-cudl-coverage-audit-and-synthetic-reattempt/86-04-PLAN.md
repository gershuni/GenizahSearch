    ```python
    """Phase 86 bridge-aware CUDL coverage scanner (NEW per Codex HIGH #3/#4).

    Walks nli_crossref.db.cambridge_manifests; for each CUDL classmark classifies
    into one of 5 tiers using shared.shelfmark_bridge.lookup_cudl AND

      - phase86_synthetic:         explain_fist_by_cudl returns 'single' AND rec.has_alma=False
                                    (new synthetic row in libraries.csv)
      - multi_inventory_ambiguous: explain_fist_by_cudl returns 'multi_inventory_ambiguous' (excluded by D-04a)
      - truly_orphan:              explain_fist_by_cudl returns 'not_found' (D-02b residue)

    Writes reports/cudl_coverage_post_phase86.csv with per-classmark classification.
    Prints summary table to stdout (consumed by Plan 04 Task 2 cudl_coverage.md authoring).

    Does NOT mutate scan_cudl_orphans.py — separate scanner per Codex HIGH #4.
    """
    from __future__ import annotations
    import csv
    import sqlite3
    import sys
    from collections import Counter
    from pathlib import Path

    from shared.shelfmark_bridge import lookup_cudl, build_alias_index
    from shared.fist_cudl_bridge import build_fist_alias_index, explain_fist_by_cudl

    ROOT = Path(__file__).resolve().parent.parent
    FIST_DB = ROOT / "fist_data" / "FIST.db"
    NLI_DB = ROOT / "nli_data" / "nli_crossref.db"
    CSV_PATH = ROOT / "libraries.csv"

    TIER_PHASE84 = "phase84_hit"
    TIER_ALIAS_ALMA = "phase86_alias_only_alma"
    TIER_SYNTHETIC = "phase86_synthetic"
    TIER_MULTI_INV = "multi_inventory_ambiguous"
    TIER_ORPHAN = "truly_orphan"


    def classify_classmark(classmark: str) -> tuple[str, str, str]:
        """Return (tier, fist_inventory_id, notes)."""
        # Tier 1: Phase 84 alias index hit.
        sys_id = lookup_cudl(classmark)
        if sys_id is not None:
            return (TIER_PHASE84, "", f"libraries.csv sys_id={sys_id}")

        # Tier 2/3/4/5: Phase 86 bridge resolution.
        status, entries = explain_fist_by_cudl(classmark)
        if status == "not_found":
            return (TIER_ORPHAN, "", "no FIST candidate via Phase 86 bridge")
        if status == "multi_inventory_ambiguous":
            inv_ids = ",".join(str(e.inventory_id) for e in entries)
            return (TIER_MULTI_INV, inv_ids, "multiple FIST InventoryIds — D-04a exclude")
        # status == 'single'
        rec = entries[0]
        if rec.has_alma:
            return (TIER_ALIAS_ALMA, str(rec.inventory_id),
                    "Covered via FIST bridge; existing Alma row — app resolution depends on Phase 84")
        return (TIER_SYNTHETIC, str(rec.inventory_id),
                "Phase 86 synthetic libraries.csv row")


    def main() -> int:
        # Build both bridge indexes (Phase 84 + Phase 86).
        # Phase 84 needs libraries.csv via build_alias_index; Phase 86 needs FIST.db.
        # We import the same csv_bank pattern used by scan_cudl_orphans.py.
        from genizah_core import csv_bank
        # Ensure csv_bank is loaded for Phase 84 build_alias_index.
        # (The exact call signature follows the existing scan_cudl_orphans.py — copy that pattern.)
        build_alias_index(csv_bank)

        fist_conn = sqlite3.connect(f"file:{FIST_DB}?mode=ro", uri=True)
        nli_conn = sqlite3.connect(f"file:{NLI_DB}?mode=ro", uri=True)
        try:
            build_fist_alias_index(fist_conn)
            cudl_rows = nli_conn.execute(
                "SELECT normalized_shelfmark FROM cambridge_manifests "
                "WHERE normalized_shelfmark IS NOT NULL AND normalized_shelfmark != '' "
                "ORDER BY normalized_shelfmark"
            ).fetchall()
        finally:
            nli_conn.close()
            # keep fist_conn open until after we write rows in case it's needed; close at end.

        rows: list[tuple[str, str, str, str]] = []
        counts: Counter[str] = Counter()
        for (classmark,) in cudl_rows:
            tier, inv_id, notes = classify_classmark(classmark)
            rows.append((classmark, tier, inv_id, notes))
            counts[tier] += 1

        fist_conn.close()

        OUT_CSV.parent.mkdir(parents=True, exist_ok=True)
        with OUT_CSV.open("w", encoding="utf-8", newline="") as f:
            writer = csv.writer(f)
            writer.writerow(["classmark", "tier", "fist_inventory_id", "notes"])
            for r in rows:
                writer.writerow(r)

        total = sum(counts.values())
        print(f"Phase 86 bridge-aware coverage scan: {total} CUDL classmarks classified.")
        print()
        print(f"{'Tier':40s}  {'Count':>8s}  {'% of total':>10s}")
        for tier in (TIER_PHASE84, TIER_ALIAS_ALMA, TIER_SYNTHETIC, TIER_MULTI_INV, TIER_ORPHAN):
            n = counts.get(tier, 0)
            pct = (100.0 * n / total) if total else 0.0
            print(f"{tier:40s}  {n:>8d}  {pct:>9.2f}%")
        print()
        print(f"Output: {OUT_CSV}")
        return 0


    if __name__ == "__main__":
        sys.exit(main())
    ```
