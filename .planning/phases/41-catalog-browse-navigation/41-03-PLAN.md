```python
class FjmsService:
    def get_domain_hierarchy(self) -> dict:
        """Dict mapping parent_domain -> {parent_domain_heb, count, children: [{domain, domain_heb, count}]}"""

    def get_browse_authors(self, domain: str = None) -> list[dict]:
        """[{"author": str, "count": int}] sorted by count desc"""

    def get_browse_works(self, domain: str = None, author: str = None) -> list[dict]:
        """[{"title": str, "title_heb": str, "count": int}] sorted by count desc"""

    def get_browse_results(self, domain: str = None, author: str = None, work: str = None, offset: int = 0, limit: int = 50) -> dict:
        """{"results": [{sys_id, title, title_heb, author, copy_date, textual_frame_heb, textual_frame_eng, domains, domains_heb}], "total": int}"""

    def get_unclassified_count(self) -> int
```
